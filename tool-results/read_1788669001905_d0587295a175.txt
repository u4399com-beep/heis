  1181→function shuffle<T>(arr: readonly T[]): T[] {
  1182→  const a = [...arr]
  1183→  for (let i = a.length - 1; i > 0; i--) {
  1184→    const j = Math.floor(Math.random() * (i + 1))
  1185→    const tmp = a[i]
  1186→    a[i] = a[j]
  1187→    a[j] = tmp
  1188→  }
  1189→  return a
  1190→}
