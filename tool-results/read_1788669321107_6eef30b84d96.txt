    47→// ---------- 浏览器指纹头组 (ff-b 增强①) ----------
    48→/**
    49→ * 场景: UA 轮换只解决了 User-Agent 单头, 真实浏览器还固定携带一组与 UA 严格配套的
    50→ * Client Hints(sec-ch-ua*)与 Fetch Metadata(Sec-Fetch-*)头 —— "只有 UA 没有配套头组"
    51→ * 与"sec-ch-ua 版本和 UA 版本不一致"/"Android UA 配桌面 platform"一样, 都是自相矛盾的
    52→ * 非浏览器指纹。本层按【实际选中的 UA】推导完整指纹头组(仅 HTTP 链注入, 见 buildHeaders):
    53→ *  - Chromium 系(Chrome/Edge): sec-ch-ua(品牌版本从 UA 提取, 与 UA 同版)+ sec-ch-ua-mobile
    54→ *    (?0/?1 按 UA 移动性)+ sec-ch-ua-platform(按 UA 平台段: Windows/macOS/Linux/Android/iOS);
    55→ *  - Sec-Fetch-*(Chromium/Firefox 导航均发送): Dest: document/Mode: navigate/User: ?1,
    56→ *    Site 按 Referer 与目标 host 关系取 none(直接输入)/same-origin(同站来源)/cross-site;
    57→ *  - Safari 不发送 Client Hints 与 Sec-Fetch-*(Fetch Metadata 不支持) → 一律不注入,
    58→ *    防"Safari UA 带 Chromium 专属头"的反向破绽; Firefox 发送 Sec-Fetch-* 但无 Client Hints;
    59→ *  - Upgrade-Insecure-Requests: 三家浏览器文档导航均发送。
    60→ * 一致性即本能力的核心: Android 移动 UA 必然配 sec-ch-ua-mobile: ?1 + platform "Android"。
    61→ * 注入点纪律: 仅 fetchHttp/fetchViaCurl(HTTP 链)经 buildHeaders({fingerprint:true}) 启用;
    62→ * 裸 Playwright 链刻意不注入 —— 真浏览器自身发送原生 Sec-Fetch-* 与 sec-ch-ua, extraHTTPHeaders
    63→ * 再注入同名头会产生重复头/覆盖冲突(双值头反而可疑), UA 之外的头组交给真浏览器自洽。
    64→ */

    74→function uaFamilyOf(ua: string): 'chromium' | 'safari' | 'firefox' | 'unknown' {
    75→  if (/Firefox\//.test(ua)) return 'firefox'
    76→  if (/Chrome\/|\bEdg\b/.test(ua)) return 'chromium'
    77→  if (/Safari\//.test(ua)) return 'safari'
    78→  return 'unknown'

    81→/** sec-ch-ua-platform 值: 从 UA 平台段推导(Chromium 系才发送, 需与 UA 自洽) */
    82→function uaPlatformHint(ua: string): string {
    83→  if (/Android/.test(ua)) return 'Android'
    84→  if (/iPhone|iPad|iOS|iPhone OS/.test(ua)) return 'iOS'
    85→  if (/Windows/.test(ua)) return 'Windows'
    86→  if (/Mac OS|Macintosh/.test(ua)) return 'macOS'
    87→  if (/CrOS/.test(ua)) return 'Chrome OS'
    88→  if (/X11|Linux/.test(ua)) return 'Linux'
    89→  return 'Windows'
    90→}

    92→/** Sec-Fetch-Site: 按 Referer 与目标 host 关系还原真实导航语义 */
    93→function secFetchSite(referer: string, targetUrl: string): string {
    94→  if (!referer) return 'none'
    95→  try {
    96→    const rHost = new URL(referer).host.toLowerCase()
    97→    const tHost = new URL(targetUrl).host.toLowerCase()
    98→    return rHost === tHost ? 'same-origin' : 'cross-site'
    99→  } catch {
   100→    return 'none'
   101→  }
   102→}

   104→/** 从 UA 生成完整指纹头组(与 buildHeaders 合并, cfg.headers 可覆盖单项) */
   105→export function fingerprintHeadersFor(ua: string, referer: string, targetUrl: string): Record<string, string> {
   106→  const family = uaFamilyOf(ua)
   107→  const headers: Record<string, string> = {
   108→    'Upgrade-Insecure-Requests': '1',
   109→  }
   110→  if (family === 'chromium' || family === 'firefox') {
   111→    headers['Sec-Fetch-Dest'] = 'document'
   112→    headers['Sec-Fetch-Mode'] = 'navigate'
   113→    headers['Sec-Fetch-Site'] = secFetchSite(referer, targetUrl)
   114→    headers['Sec-Fetch-User'] = '?1'
   115→  }
   116→  if (family === 'chromium') {
   117→    const cv = ua.match(CHROME_VER_RE)?.[1] || ''
   118→    const ev = ua.match(EDGE_VER_RE)?.[1] || ''
   119→    if (cv) {
   120→      const brands = ev
   121→        ? `"Chromium";v="${cv}", "Google Chrome";v="${cv}", "Microsoft Edge";v="${ev}", "Not:A-Brand";v="24"`
   122→        : `"Chromium";v="${cv}", "Google Chrome";v="${cv}", "Not:A-Brand";v="24"`
   123→      headers['sec-ch-ua'] = brands
   124→      headers['sec-ch-ua-mobile'] = isMobileUa(ua) ? '?1' : '?0'
   125→      headers['sec-ch-ua-platform'] = `"${uaPlatformHint(ua)}"`
   126→    }
   127→  }
   128→  return headers
   129→}

   132→/** 会话条目 TTL(ff-b 增强④): 挑战/会话 Cookie(如 cf_clearance)与出口 IP+UA 绑定,
   133→ *  30 分钟前的陈旧会话继续携带反而是"过期会话+拒绝服务"的 403 诱因 —— 真实浏览器
   134→ *  会话同样有时效。过期条目在 get/count 惰性清扫; 跨请求复用本体(按 host 缓存、
   135→ *  下次同 host 直接带)是既有 autoCookie 全局罐能力, 本轮仅补时效与失效清理 */
   136→const COOKIE_SESSION_TTL_MS = 30 * 60 * 1000

   138→class CookieJar {
   139→  private jars = new Map<string, Map<string, { v: string; at: number }>>()
   140→  /** 未过期条目判定(过期即惰性删除) */
   141→  private fresh(jar: Map<string, { v: string; at: number }>, k: string, e: { at: number }): boolean {
   142→    if (Date.now() - e.at < COOKIE_SESSION_TTL_MS) return true
   143→    jar.delete(k)
   144→    return false
   145→  }
   146→  get(domain: string): string {
   147→    const jar = this.jars.get(domain)
   148→    if (!jar || jar.size === 0) return ''
   149→    const out: string[] = []
   150→    for (const [k, e] of jar) {
   151→      if (this.fresh(jar, k, e)) out.push(`${k}=${e.v}`)
   152→    }
   153→    return out.join('; ')
   154→  }
   155→  /** 当前域名已存(未过期)cookie 数(用于判断本次响应是否刚种下新 Cookie) */
   156→  count(domain: string): number {
   157→    const jar = this.jars.get(domain)
   158→    if (!jar) return 0
   159→    let n = 0
   160→    for (const [k, e] of jar) {
   161→      if (this.fresh(jar, k, e)) n++
   162→    }
   163→    return n
   164→  }
   165→  store(domain: string, setCookieHeaders: string[]) {
   166→    if (!setCookieHeaders?.length) return
   167→    let jar = this.jars.get(domain)
   168→    if (!jar) { jar = new Map(); this.jars.set(domain, jar) }
   169→    for (const raw of setCookieHeaders) {
   170→      const [pair] = raw.split(';')
   171→      const idx = pair.indexOf('=')
   172→      if (idx > 0) jar.set(pair.slice(0, idx).trim(), { v: pair.slice(idx + 1).trim(), at: Date.now() })
   173→    }
   174→  }
   175→  seed(domain: string, cookieStr?: string) {
   176→    if (!cookieStr) return
   177→    let jar = this.jars.get(domain)
   178→    if (!jar) { jar = new Map(); this.jars.set(domain, jar) }
   179→    for (const pair of cookieStr.split(';')) {
   180→      const idx = pair.indexOf('=')
   181→      if (idx > 0) jar.set(pair.slice(0, idx).trim(), { v: pair.slice(idx + 1).trim(), at: Date.now() })
   182→    }
   183→  }
   184→  /** 清空指定 host 的罐(ff-b): 403 且无新 Cookie 时疑陈旧会话, 清空重走 autoCookie */
   185→  clear(domain: string) {
   186→    this.jars.delete(domain)
   187→  }
   188→}

   201→function stripBom(s: string): string {
   202→  // 去 UTF-8 BOM(\uFEFF): 避免首段隐形字符污染标题匹配/正文开头
   203→  return s.charCodeAt(0) === 0xfeff ? s.slice(1) : s
   204→}

   206→function decodeBuffer(buf: ArrayBuffer, contentType?: string): string {
   207→  let charset = ''
   208→  const ct = contentType || ''
   209→  // 兼容 charset="gb2312" / charset='gbk' 引号变体(原正则遇到引号即失配, 编码退化为 utf8 产生乱码)
   210→  const m1 = ct.match(/charset\s*=\s*["']?([\w-]+)/i)
   211→  if (m1) charset = m1[1]
   212→  const head = Buffer.from(buf.slice(0, 2048))
   213→  if (!charset) {
   214→    const headStr = head.toString('latin1')
   215→    const m2 = headStr.match(/<meta[^>]+charset=["']?([\w-]+)/i)
   216→    if (m2) charset = m2[1]
   217→    else if (headStr.includes('charset=gb')) charset = 'gbk'
   218→  }
   219→  charset = charset.toLowerCase()
   220→  if (!charset || charset === 'utf-8' || charset === 'utf8') {
   221→    try { return stripBom(new TextDecoder('utf-8', { fatal: false }).decode(buf)) } catch { return stripBom(Buffer.from(buf).toString('utf8')) }
   222→  }
   223→  // encodingExists 兜底: 非法/未知名编码(如 x-mac-cyrillic)退回 utf8, 不让 iconv 抛错
   224→  if (iconv.encodingExists(charset)) {
   225→    return stripBom(iconv.decode(Buffer.from(buf), charset))
   226→  }
   227→  return stripBom(Buffer.from(buf).toString('utf8'))
   228→}

   230→// ---------- JS 跳转挑战识别 ----------
   231→/**
   232→ * 判定"JS跳转挑战壳"页:
   233→ *  - 内容 <1200 字 且 含 window.location / location.href / location.replace 等跳转脚本
   234→ *  - 或含 http-equiv="refresh" 且内容很短(<1200)
   235→ * 典型: 反爬中间页只输出一段脚本跳到真实地址 / 首次访问种 Cookie 后刷新
   236→ */
   237→export function isJsChallenge(html: string): boolean {
   238→  if (!html) return false
   239→  const s = html.trim()
   240→  if (!s || s.length >= 1200) return false
   241→  const hasRedirect = /window\.location\s*[.[]|location\.href\s*=|location\.replace\s*\(|location\.assign\s*\(/.test(s)
   242→  const hasRefresh = /http-equiv\s*=\s*["']?refresh/i.test(s)
   243→  return hasRedirect || hasRefresh
   244→}

   266→function hasNormalTitle(html: string): boolean {
   267→  const m = html.match(/<title[^>]*>([\s\S]{1,300}?)<\/title>/i)
   268→  if (!m) return false
   269→  const t = m[1].trim().toLowerCase()
   270→  if (t.length < 2) return false
   271→  // 修复: 补 '请稍候/请稍後'(CF 中文盾页标题)——原先这类标题被当正常标题豁免, 盾页被当正文
   272→  const bad = ['just a moment', 'attention required', 'access denied', 'forbidden', '请开启', '验证', '请稍候', '请稍後', '403', '404']
   273→  return !bad.some((k) => t.includes(k))
   274→}

   276→export function looksBlocked(html: string): boolean {
   277→  if (!html) return true
   278→  // 用 isJsChallenge 区分: 极短 JS 跳转壳直接判拦
   279→  if (isJsChallenge(html)) return true
   280→  const lower = html.toLowerCase()
   281→  // CF JS Detections 脚本(challenge-platform/scripts/jsd/main.js)是 Bot Management 下
   282→  // 正常页面普遍内嵌的探测脚本, 不代表当前是挑战页 —— 页面有正常标题且足够长时豁免,
   283→  // 否则真实内容页(101kks 实测)被永久拒收
   284→  const jsdBenign = lower.includes('challenge-platform/scripts/jsd') && html.length >= 1200 && hasNormalTitle(html)
   285→  // 强挑战特征(CF 等): 无论长短一律判拦
   286→  if (!jsdBenign && STRONG_BLOCK_MARKERS.some((k) => lower.includes(k))) return true
   287→  // 保留原规则: 极短内容视为被拦
   288→  if (html.length < 200) return true
   289→  // 含正常 <title> 且长度 >= 1200 的页面视为正常内容页,
   290→  // 不因正文/导航提及"验证码/verify/enable javascript"等词误判
   291→  if (html.length >= 1200 && hasNormalTitle(html)) return false
   292→  return BLOCK_MARKERS.some((k) => lower.slice(0, 4000).includes(k))
   293→}

   308→function pickUaFor(domain: string, cfg: FetchConfig): string {
   309→  if (cfg.uaMode === 'custom' && cfg.customUa) return cfg.customUa
   310→  const pinned = domain ? domainUa.get(domain) : undefined
   311→  if (pinned) {
   312→    // ff-b uaMode=mobile/desktop: 钉扎 UA 与请求类别不符(模式在线切换)时重选,
   313→    // 防止"desktop 模式拿到上一次 mobile 钉扎的 iPhone UA"自相矛盾
   314→    if (cfg.uaMode !== 'mobile' && cfg.uaMode !== 'desktop') return pinned
   315→    if (isMobileUa(pinned) === (cfg.uaMode === 'mobile')) return pinned
   316→  }
   317→  // ff-b: mobile/desktop 两档 = 池内移动/桌面子集随机(同域钉扎语义不变);
   318→  // rotate/fixed 维持全池随机
   319→  let ua: string
   320→  if (cfg.uaMode === 'mobile' || cfg.uaMode === 'desktop') {
   321→    const subset = UA_POOL.filter((u) => isMobileUa(u) === (cfg.uaMode === 'mobile'))
   322→    ua = subset.length ? subset[Math.floor(Math.random() * subset.length)] : randomUa()
   323→  } else {
   324→    ua = randomUa()
   325→  }
   326→  if (domain) {
   327→    if (domainUa.size > 200) domainUa.clear() // 防站群场景无限增长
   328→    domainUa.set(domain, ua)
   329→  }
   330→  return ua
   331→}

   384→ *  context 覆盖的前提, Playwright 文档: 所有 context 覆盖后全局值永不使用, 可为任意串)
   385→ *  + newContext 注入真实代理(per-context, 本浏览器实例仅本请求专用, 无共享串扰);

   387→async function renderWithBrowserRaw(url: string, cfg: FetchConfig, ua: string): Promise<string> {
   388→  if (!pwModule) {
   389→    const ok = await checkBrowser()
   390→    if (!ok) throw new Error('浏览器渲染引擎不可用(未安装playwright/chromium), 请使用HTTP引擎')
   391→  }
   392→  const { chromium } = pwModule
   393→  const proxy = pickProxyFor(url, cfg)
   394→  const browser = await chromium.launch({
   395→    headless: true,
   396→    args: ['--no-sandbox', '--disable-blink-features=AutomationControlled'],
   397→    ...(proxy ? { proxy: { server: 'http://per-context-placeholder' } } : {}),
   398→  })
   399→  const timeoutMs = cfg.timeout && cfg.timeout > 0 ? cfg.timeout : 20000
   400→  try {
   401→    const ctx = await browser.newContext({
   402→      userAgent: ua,
   403→      viewport: { width: 1366, height: 768 },
   404→      extraHTTPHeaders: buildHeaders(url, cfg, ua),
   405→      ...(proxy ? { proxy: playwrightProxyParts(proxy) } : {}),
   406→    })
   407→    // 反自动化检测脚本
   408→    await ctx.addInitScript(() => {
   409→      Object.defineProperty(navigator, 'webdriver', { get: () => undefined })

   413→    await ctx.addInitScript(buildIdentityInitScript(ua))
   414→    const page = await ctx.newPage()
   415→    // hh-d2: CDP Network.setUserAgentOverride(+userAgentMetadata) 让网络层 sec-ch-ua* 头与
   416→    // UA 字符串三方自洽(含移动分支); 与 Obscura 同一 helper, 失败容忍(JS 面仍有身份脚本)
   417→    await applyUaCdpOverride(page, ua)
   418→    if (cfg.waitSelector) {
   419→      await page.goto(url, { waitUntil: 'domcontentloaded', timeout: timeoutMs })
   420→      try { await page.waitForSelector(cfg.waitSelector, { timeout: cfg.waitMs || 8000 }) } catch { /* 容忍 */ }
   421→    } else {
   422→      await page.goto(url, { waitUntil: 'networkidle', timeout: timeoutMs }).catch(async () => {
   423→        await page.goto(url, { waitUntil: 'domcontentloaded', timeout: timeoutMs })
   424→      })
   425→    }
   426→    if (cfg.waitMs) await page.waitForTimeout(cfg.waitMs)
   427→    // 点击展开懒加载内容(与 Obscura 路径对齐, 裸 Playwright 降级路径同样支持);
   428→    // gg: 主 frame+跨域 iframe 全遍历(挑战复选框在跨域 iframe 内, dd-d 缺口补齐),
   429→    // 找不到元素静默跳过语义不变
   430→    if (cfg.clickSelector) {
   431→      const clicked = await clickSelectorAnywhere(page, cfg.clickSelector)
   432→      if (clicked) await page.waitForTimeout(1200)
   433→    }
   434→    let html: string
   435→    try {
   436→      html = await page.content()
   437→    } catch {
   438→      // 点击"展开"可能实为链接触发整页导航: content() 在导航提交期间会抛错,
   439→      // 退避后重试一次拿导航后的真实内容, 仍失败才向上抛
   440→      await page.waitForTimeout(1500)
   441→      html = await page.content()
   442→    }
   443→    await ctx.close()
   444→    return html
   445→  } finally {
   446→    await browser.close()
   447→  }
   448→}

   450→/** 头组构造(HTTP 内容链专用; ff-b 增强①: opts.fingerprint=true 时注入完整浏览器指纹头组)
   451→ *  - 指纹纪律: 仅 fetchHttp(逐跳)/fetchViaCurl 传入 fingerprint —— 裸 Playwright 链
   452→ *    (renderWithBrowser)与 fetchBinary 刻意不传: 真浏览器自发自洽的原生 sec-ch-ua/Sec-Fetch-*,
   453→ *    再注入同名头会产生重复/冲突(双值头反而可疑); 资源请求的 Sec-Fetch-Dest 语义也不同
   454→ *  - refererChain(ff-b 增强②): cfg.refererChain && cfg.refererUrl 时 Referer 用运行时注入的
   455→ *    来源页 URL(目录页→书籍页→章节页同链路), 未注入回退站点 origin(零回归)
   456→ *  - 合并次序: 基础头 → 指纹头组 → cfg.headers(规则显式配置最优先, 可覆盖任意单项) */
   457→function buildHeaders(url: string, cfg: FetchConfig, ua: string, opts?: { fingerprint?: boolean }): Record<string, string> {
   458→  let origin = ''
   459→  try { origin = new URL(url).origin } catch { /* ignore */ }
   460→  const headers: Record<string, string> = {
   461→    'User-Agent': ua,
   462→    Accept: 'text/html,application/xhtml+xml,application/xml;q=0.9,image/avif,image/webp,*/*;q=0.8',
   463→    'Accept-Language': 'zh-CN,zh;q=0.9,en;q=0.6',
   464→    'Cache-Control': 'no-cache',
   465→  }
   466→  const chainReferer = cfg.refererChain && cfg.refererUrl ? cfg.refererUrl : ''
   467→  if (opts?.fingerprint) {
   468→    // 指纹头组按【实际选中 UA】+【生效 Referer】推导(Sec-Fetch-Site 语义依赖后者);
   469→    // 先于 cfg.headers 合并 —— 规则显式配置的头永远最优先
   470→    Object.assign(headers, fingerprintHeadersFor(ua, chainReferer || origin, url))
   471→  }
   472→  Object.assign(headers, cfg.headers)
   473→  if (chainReferer) headers.Referer = chainReferer
   474→  else if (cfg.referer !== false && origin) headers.Referer = origin
   475→  // Cookie 合并去重: 同名键以罐中值(服务端最新 Set-Cookie)为准, 避免拼出 "a=1; a=9" 重复 Cookie 头
   476→  const merged = new Map<string, string>()
   477→  for (const src of [cfg.cookies, cookieJar.get(originHost(url))]) {
   478→    if (!src) continue
   479→    for (const pair of src.split(';')) {
   480→      const idx = pair.indexOf('=')
   481→      if (idx > 0) merged.set(pair.slice(0, idx).trim(), pair.slice(idx + 1).trim())
   482→    }
   483→  }
   484→  if (merged.size) headers.Cookie = Array.from(merged.entries()).map(([k, v]) => `${k}=${v}`).join('; ')
   485→  return headers
   486→}

   517→/** 代理条目形态校验: scheme 白名单(http/https/socks5(h)/socks4(a)) + 无空白/逗号的
   518→ *  host[:port] 形态(凭证以 http://u:p@host:port 内联), 单条 ≤500 字符。
   519→ *  与 types.ts sanitizeFetchConfig 内联校验同口径(两处保持一致, 改动需同步) */
   520→export function isValidProxySpec(s: string): boolean {
   521→  return s.length <= 500 && /^(https?|socks5h?|socks4a?):\/\/[^\s,]+$/.test(s)
   522→}

   524→/** 代理池解析: 逗号分隔多条, 去空/去重/逐条校验, 上限 MAX_PROXY_POOL */
   525→export function parseProxyPool(proxyUrl: string | undefined | null): string[] {
   526→  if (!proxyUrl) return []
   527→  const seen = new Set<string>()
   528→  const out: string[] = []
   529→  for (const raw of proxyUrl.split(',')) {
   530→    const s = raw.trim()
   531→    if (!s || seen.has(s) || !isValidProxySpec(s) || out.length >= MAX_PROXY_POOL) continue
   532→    seen.add(s)
   533→    out.push(s)
   534→  }
   535→  return out
   536→}

   538→/** 回环豁免: 目标 host 为 localhost/*.localhost/127.0.0.0/8/::1/0.0.0.0 时跳过代理直连
   539→ *  —— 否则本地 mock 服务/token 代理 tokenUrl(如 bqg713-proxy 127.0.0.1:3010)会被代理
   540→ *  转发出不去。hostname 对 IPv6 含方括号需剥离 */
   541→export function isLoopbackTarget(url: string): boolean {
   542→  try {
   543→    const h = new URL(url).hostname.toLowerCase().replace(/^\[|\]$/g, '')
   544→    return h === 'localhost' || h.endsWith('.localhost') || h === '::1' || h === '0.0.0.0' || /^127\.\d{1,3}\.\d{1,3}\.\d{1,3}$/.test(h)
   545→  } catch {
   546→    return false
   547→  }
   548→}

   550→/**
   551→ * 代理选路(三链路单一收敛点, 返回本次请求使用的代理, ''=直连):
   552→ * - 未配置 / 目标回环 → 直连
   553→ * - 配置多条 → 均匀随机轮换(与 UA 池同款 random 模式, 分布可测试验证)
   554→ * fetchHttp(bun fetch)/fetchViaCurl(curl)/renderWithBrowserRaw(per-context)一律经
   555→ * 本函数取代理, 避免三处重复实现漂移
   556→ */
   557→export function pickProxyFor(url: string, cfg: FetchConfig): string {
   558→  const pool = parseProxyPool(cfg.proxyUrl)
   559→  if (!pool.length || isLoopbackTarget(url)) return ''
   560→  return pool[Math.floor(Math.random() * pool.length)]
   561→}

   568→/** Playwright per-context proxy 参数: 内联凭证拆出 username/password
   569→ *  (Playwright 不接受 server 内嵌凭证), 无凭证原样返回 */
   570→function playwrightProxyParts(proxy: string): { server: string; username?: string; password?: string } {
   571→  try {
   572→    const u = new URL(proxy)
   573→    if (u.username || u.password) {
   574→      const out: { server: string; username?: string; password?: string } = {
   575→        server: `${u.protocol}//${u.host}`,
   576→      }
   577→      const un = decodeURIComponent(u.username)
   578→      const pw = decodeURIComponent(u.password)
   579→      if (un) out.username = un
   580→      if (pw) out.password = pw
   581→      return out
   582→    }
   583→  } catch { /* 已过 isValidProxySpec, 理论不达 */ }
   584→  return { server: proxy }

   592→// ---------- Retry-After 头抢救 (ab-b: 429 限流冷却精确感知) ----------
   593→/**
   594→ * 场景(zz-b 遗留收编): 真 429 在 HTTP 引擎以抛错形态抵达 runner.gateFetch, 但抛错对象
   595→ * 原先只保留 status/bodyHtml, Retry-After 头在抛错瞬间丢失 → 限流冷却一律走 30s 兜底。
   596→ * 现把解析出的毫秒值挂到抛错对象新字段 retryAfterMs(既有 status/bodyHtml 行为零变化),
   597→ * runner 抛错路径透传给 reportHostRateLimited —— 服务端给多少歇多久(上限/噪声底由
   598→ * hostgate 侧钳制), 不再盲目硬等 30s。
   599→ * 解析语义(RFC 7231 Retry-After 两种形态):
   600→ *  - 整数秒: '2' → 2000('0' 如实返回 0, <1s 噪声底由 hostgate 兜底 30s);
   601→ *  - HTTP 日期: Date.parse 兜底, 取"距现时刻"毫秒(已过期返回 0);
   602→ *  - 缺失/空/垃圾 → undefined(调用方不挂字段, 上层走 30s 兜底)。
   603→ * 导出供验证脚本直接单测解析语义(verify-ab-b-ratelimit)
   604→ */
   605→export function parseRetryAfterHeaderMs(raw: string | null | undefined): number | undefined {
   606→  const s = (raw ?? '').trim()
   607→  if (!s) return undefined
   608→  if (/^\d+$/.test(s)) {
   609→    const sec = parseInt(s, 10)
   610→    return Number.isSafeInteger(sec) ? sec * 1000 : undefined
   611→  }
   612→  const t = Date.parse(s)
   613→  if (!Number.isFinite(t)) return undefined
   614→  return Math.max(0, t - Date.now())
   615→}

   617→/** 把 Retry-After 毫秒值挂到 HTTP 抛错对象(ab-b): 头缺失/解析失败时不挂字段
   618→ *  (err.retryAfterMs 保持 undefined), 既有 err.status/err.bodyHtml 行为完全不变。
   619→ *  res.headers 兼容 native fetch 与 gg 中继重组形态(均为 Headers 实例) */
   620→function attachRetryAfterMs(err: any, headers: { get(name: string): string | null }): void {
   621→  const ms = parseRetryAfterHeaderMs(headers?.get ? headers.get('retry-after') : null)
   622→  if (ms !== undefined) err.retryAfterMs = ms
   623→}

   625→async function fetchHttp(url: string, cfg: FetchConfig, ua: string, proxy = '', transport: 'native' | 'relay' = 'native'): Promise<string> {
   626→  // 超时防御: 规则配置里 timeout 可能是 0/null/负数, setTimeout(fn, 0) 会立即中止请求
   627→  const timeoutMs = cfg.timeout && cfg.timeout > 0 ? cfg.timeout : 20000
   628→  const controller = new AbortController()
   629→  // ee-d: 本计时器是“超时型 AbortError”的唯一来源(stop 不中止在途——abortControllers 声明后从未使用),
   630→  // 打标后 runner/gateFetch 可区分“源站超时”与“停止/换代在途中止”, 前者计失败嗂 hostGate, 后者才享 x-a 豁免
   631→  let timedOut = false
   632→  const timer = setTimeout(() => { timedOut = true; controller.abort() }, timeoutMs)
   633→  // 出口代理(dd-a): ''=直连; bun fetch 原生 RequestInit.proxy 仅支持 http/https,
   634→  // socks5 条目在此链即时失败(UnsupportedProxyProtocol)后由 fetchHttpWithCurlFallback
   635→  // 同代理重试 curl 链(-x 全形态), 支持矩阵见代理池段注释
   636→  try {
   637→    // redirect:'manual' 自循环逐跳收集(y-a增强重放, 补 x-a 遗留②): 原先 redirect:'follow'
   638→    // 只能拿到【最终响应】的 Set-Cookie —— 多跳重定向中, 经 301/302 中间跳种会话 Cookie
   639→    // 的站(首访种 Cookie 再跳真实页)中间跳 Cookie 全部丢失, 同站 http→https 升级链路
   640→    // https 侧永远拿不到会话 Cookie(x-a 修复只解决了"归属域", 没解决"只收最终一跳")。
   641→    // 改为逐跳: 每跳 Set-Cookie 归属到【该跳实际 URL】的域键(与 curl 路径按轮归属同语义),
   642→    // 后续跳经 buildHeaders 带上前面跳种下的 Cookie —— 等价于真实浏览器跟随重定向的
   643→    // Cookie 行为。相对 Location 解析; 20 跳上限; 跨 scheme 降级(https→http)拒绝,
   644→    // http→https 升级放行(国内站 301 升级 https 常态, 不能因安全策略拒采)。
   645→    // 注: Bun fetch redirect:'manual' 实测(1.3.14)返回真实 3xx 响应, 状态行/Location/
   646→    // getSetCookie 全可读, 无 opaque-redirect 屏蔽(见 scripts/archive/probe-bun-manual-redirect.ts)
   647→    let hopUrl = url
   648→    for (let hop = 0; ; hop++) {
   649→      if (hop > MAX_REDIRECT_HOPS) {
   650→        throw new Error(`HTTP 重定向超过 ${MAX_REDIRECT_HOPS} 跳上限(疑似重定向环)`)
   651→      }
   652→      // ff-b①: HTTP 内容链逐跳注入完整指纹头组(与 UA 自洽的 sec-ch-ua*/Sec-Fetch-*)
   653→      const headers = buildHeaders(hopUrl, cfg, ua, { fingerprint: true })
   654→      // 出口代理逐跳同代理(会话连贯性/出口固定); 交叉类型携带非标准 proxy 字段
   655→      // (Bun 运行时扩展生效, 不依赖 bun-types 全局声明)
   656→      const init: RequestInit & { proxy?: string } = { headers, redirect: 'manual', signal: controller.signal }
   657→      if (proxy) init.proxy = proxy
   658→      // gg 中继桥: transport='relay' 时逐跳经 bun 中继服务发起(响应重组为 Response 形态,
   659→      // status/location/getSetCookie/arrayBuffer 全保持 —— 逐跳重定向/Cookie 收集/超时/
   660→      // 指纹头组语义全部复用本循环, 与 native 传输唯一差异在底层传输介质)
   661→      const res = transport === 'relay' && proxy
   662→        ? await relayHop(hopUrl, headers, proxy, controller.signal, timeoutMs)
   663→        : await fetch(hopUrl, init)
   664→      if (cfg.autoCookie !== false) {
   665→        const setCookies = typeof res.headers.getSetCookie === 'function' ? res.headers.getSetCookie() : []
   666→        // 每跳 Cookie 记到该跳 URL 的 origin 名下: 跨域重定向不串味, 同站跳转按域聚合
   667→        cookieJar.store(originHost(hopUrl), setCookies)
   668→      }
   669→      const location = res.headers.get('location')
   670→      if (res.status >= 300 && res.status < 400 && location) {
   671→        // rr-c3 卫生: manual 重定向链上 3xx 响应体从不读取, 显式 cancel 立即释放连接
   672→        // (不消费的 body 由 GC 延迟回收, 重定向密集站滞后占用连接池; 可选链短路安全,
   673→        //  relay 重组形态无 body 字段时为 no-op; catch 兜底防空 rejection)
   674→        try { void res.body?.cancel().catch(() => {}) } catch { /* ignore */ }
   675→        let next: URL
   676→        try {
   677→          next = new URL(location, hopUrl) // 相对 Location(./x、/x、//host)按当前跳解析
   678→        } catch {
   679→          // 非法 Location: 视作最终响应走 !res.ok 抛错语义(带 status+bodyHtml)
   680→          // ee-d: 错误体同样走 charset 感知解码(GBK 站挑战壳若按 utf8 读成 FFFD, looksBlocked/isJsChallenge 全部漏判)
   681→          const bodyHtml = await res.arrayBuffer().then((b) => decodeBuffer(b, res.headers.get('content-type') ?? undefined)).catch(() => '')
   682→          const err: any = new Error(`HTTP ${res.status}(Location 非法)`)
   683→          err.status = res.status
   684→          err.bodyHtml = bodyHtml
   685→          attachRetryAfterMs(err, res.headers) // ab-b: 有 res 在手, 错误形态统一抢救 Retry-After
   686→          throw err
   687→        }
   688→        if (next.protocol !== new URL(hopUrl).protocol) {
   689→          // 跨 scheme: 仅放行 http→https 升级; 降级(https→http)与其余一律拒绝,
   690→          // 防止降级明文跳转把会话 Cookie 带到不安全上下文
   691→          const upgrade = new URL(hopUrl).protocol === 'http:' && next.protocol === 'https:'
   692→          if (!upgrade) {
   693→            const err: any = new Error(`HTTP ${res.status} 重定向跨 scheme 被拒绝(${new URL(hopUrl).protocol}→${next.protocol})`)
   694→            err.status = res.status
   695→            attachRetryAfterMs(err, res.headers) // ab-b: 同上(3xx 错误形态, 头在才挂)
   696→            throw err
   697→          }
   698→        }
   699→        hopUrl = next.toString()
   700→        continue
   701→      }
   702→      if (!res.ok) {
   703→        // 读出错误响应体供挑战识别(isJsChallenge/CF壳), 挂在 error.bodyHtml 上
   704→        // ee-d: 与成功路径同走 decodeBuffer(charset 三级探测), 否则 GBK 站 403 壳页乱码化后挑战识别失效
   705→        const bodyHtml = await res.arrayBuffer().then((b) => decodeBuffer(b, res.headers.get('content-type') ?? undefined)).catch(() => '')
   706→        const err: any = new Error(`HTTP ${res.status}`)
   707→        err.status = res.status
   708→        err.bodyHtml = bodyHtml
   709→        // ab-b(429 主通道): 真 429 以抛错形态抵达 runner.gateFetch —— 此处是 Retry-After
   710→        // 头唯一能被抢救的位置(zz-b 遗留: 原先头信息在此丢失, 限流冷却一律 30s 兜底)
   711→        attachRetryAfterMs(err, res.headers)
   712→        throw err
   713→      }
   714→      const buf = await res.arrayBuffer()
   715→      return decodeBuffer(buf, res.headers.get("content-type") ?? undefined)
   716→    }
   717→  } catch (e: any) {
   718→    // ee-d: fetch 超时(本计时器 abort)打标 isFetchTimeout —— 上层据此分类为源站超时
   719→    // (计 errors+写日志+嗂 hostGate 连败), 不再被 x-a 停止豁免分支静默吞掉
   720→    if ((e?.name === 'AbortError' || e?.code === 'ABORT_ERR') && timedOut) e.isFetchTimeout = true
   721→    throw e
   722→  } finally {
   723→    clearTimeout(timer)
   724→  }
   725→}

   727→// ---------- curl 子进程传输(反 TLS 指纹封锁) ----------
   728→/**
   729→ * 场景: 部分站点的 WAF/CDN(uukanshu.cc 实测)按 TLS 指纹(JA3)封锁常见 HTTP 客户端 ——
   730→ * Bun/Node fetch(BoringSSL 栈)必被 403, 而系统 curl(OpenSSL 栈)可直连。
   731→ * 故在 HTTP 引擎内提供第二级传输: curl 子进程。
   732→ * - argv 数组式 spawn(不经 shell, 无注入面), 仅支持 http/https
   733→ * - 复用 CookieJar(autoCookie)与 buildHeaders(UA轮换/Referer)
   734→ * - 响应头落临时文件(-D), body 走 stdout(二进制安全), 10MB 上限防内存炸
   735→ */
   736→let curlAvailable: boolean | null = null
   737→let curlCheckedAt = 0
   738→const CURL_PROBE_RETRY_MS = 60_000
   739→
   740→/** Buffer(视图) -> 独立 ArrayBuffer(拷贝, 防共享池越界) */
   741→function toArrayBufferView(buf: Buffer): ArrayBuffer {
   742→  return buf.buffer.slice(buf.byteOffset, buf.byteOffset + buf.byteLength) as ArrayBuffer
   743→}
   744→
   745→async function checkCurl(): Promise<boolean> {
   746→  if (curlAvailable === true) return true
   747→  if (curlAvailable === false && Date.now() - curlCheckedAt < CURL_PROBE_RETRY_MS) return false
   748→  try {
   749→    const { spawn } = await import('node:child_process')
   750→    curlAvailable = await new Promise<boolean>((resolve) => {
   751→      const child = spawn('curl', ['--version'], { stdio: ['ignore', 'ignore', 'ignore'] })
   752→      const t = setTimeout(() => {
   753→        try { child.kill() } catch { /* ignore */ }
   754→        resolve(false)
   755→      }, 5000)
   756→      child.on('error', () => { clearTimeout(t); resolve(false) })
   757→      child.on('close', (code) => { clearTimeout(t); resolve(code === 0) })
   758→    })
   759→  } catch {
   760→    curlAvailable = false
   761→  }
   762→  curlCheckedAt = Date.now()
   763→  return curlAvailable
   764→}
   765→
   766→/** curl 子进程传输(内部实现, 导出仅供诊断/冒烟脚本直接复用)
   767→ *  proxy(dd-a): 非空时以 -x 透传(http/https/socks5(h)/socks4(a) 全形态, 内联凭证
   768→ *  http://u:p@host:port 原生支持; 值清洗控制字符防 curl 参数注入) */
   769→export async function fetchViaCurl(url: string, cfg: FetchConfig, ua: string, proxy = ''): Promise<string> {
   770→  if (!/^https?:\/\//i.test(url)) throw new Error('curl 传输仅支持 http/https URL')
   771→  if (!(await checkCurl())) throw new Error('curl 子进程不可用')
   772→  // ff-b①: curl 子进程同为 HTTP 内容链, 指纹头组与 bun fetch 链同口径(双传输一致指纹, 防降级后头组消失露馅)
   773→  const headers = buildHeaders(url, cfg, ua, { fingerprint: true })
   774→  const timeoutMs = cfg.timeout && cfg.timeout > 0 ? cfg.timeout : 20000
   775→  const [{ tmpdir }, { join }, { randomUUID }] = await Promise.all([
   776→    import('node:os'), import('node:path'), import('node:crypto'),
   777→  ])
   778→  const headerFile = join(tmpdir(), `novel-curl-${randomUUID()}.hdr`)
   779→  const args: string[] = [
   780→    '-sS', '-L', '--max-redirs', '5', '--compressed',
   781→    '--max-time', String(Math.max(2, Math.ceil(timeoutMs / 1000))),
   782→    '-D', headerFile, '-o', '-',
   783→  ]
   784→  for (const [k, v] of Object.entries(headers)) {
   785→    // 控制字符清洗: 防 header 值/键换行注入额外 curl 指令(argv 传输仍单参数, 但 curl 自身按行解析);
   786→    // 键同样要洗(键来自 cfg.headers 用户配置), 且去冒号防 curl 把键值对解析错位
   787→    const key = String(k).replace(/[\r\n\0:]+/g, '').trim()
   788→    const clean = String(v).replace(/[\r\n\0]+/g, ' ').trim()
   789→    if (key && clean) args.push('-H', `${key}: ${clean}`)
   790→  }
   791→  if (proxy) {
   792→    // 出口代理: -x 全形态; 控制字符清洗与头注入同口径(curl 按行解析参数值)
   793→    args.push('-x', proxy.replace(/[\r\n\0]+/g, ''))
   794→  }
   795→  args.push('--', url)
   796→  const { spawn } = await import('node:child_process')
   797→  const { readFile, unlink } = await import('node:fs/promises')
   798→
   799→  return await new Promise<string>((resolve, reject) => {
   800→    const child = spawn('curl', args, { stdio: ['ignore', 'pipe', 'pipe'] })
   801→    const chunks: Buffer[] = []
   802→    let total = 0
   803→    const MAX_HTML_BYTES = 10 * 1024 * 1024
   804→    let stderr = ''
   805→    let settled = false
   806→    // 溢出标记: 原实现超限 SIGKILL 后 close 处理器仍会把已收到的部分 body 当成功内容 resolve,
   807→    // 截断 HTML 会被上层解析成半截正文/目录入库 —— 必须改为 reject
   808→    let overflow = false
   809→    const killTimer = setTimeout(() => {
   810→      try { child.kill('SIGKILL') } catch { /* ignore */ }
   811→    }, timeoutMs + 5000)
   812→    child.stdout.on('data', (c: Buffer) => {
   813→      if (overflow) return
   814→      total += c.length
   815→      if (total > MAX_HTML_BYTES) {
   816→        overflow = true
   817→        chunks.length = 0 // 立即释放已收数据, 不留大块缓冲到 close
   818→        try { child.kill('SIGKILL') } catch { /* ignore */ }
   819→        return
   820→      }
   821→      chunks.push(c)
   822→    })
   823→    child.stderr.on('data', (c: Buffer) => {
   824→      if (stderr.length < 600) stderr += c.toString()
   825→    })
   826→    child.on('error', (e) => {
   827→      if (settled) return
   828→      settled = true
   829→      clearTimeout(killTimer)
   830→      // 修复: error 路径(spawn 后期失败/进程无法被 kill)原先不删临时头文件, 造成 tmp 泄漏
   831→      try { void unlink(headerFile).catch(() => {}) } catch { /* ignore */ }
   832→      reject(e)
   833→    })
   834→    child.on('close', () => {
   835→      if (settled) return
   836→      settled = true
   837→      clearTimeout(killTimer)
   838→      ;(async () => {
   839→        let headerRaw = ''
   840→        try {
   841→          headerRaw = await readFile(headerFile, 'utf8')
   842→        } catch { /* 无头文件: curl 提前失败 */ }
   843→        try { await unlink(headerFile) } catch { /* ignore */ }
   844→        const body = Buffer.concat(chunks)
   845→        // 修复: 10MB 上限溢出后(SIGKILL)原先仍会走到 resolve 返回截断内容
   846→        if (overflow) {
   847→          reject(new Error(`curl 响应体超过 ${Math.round(MAX_HTML_BYTES / 1024 / 1024)}MB 上限, 已中止`))
   848→          return
   849→        }
   850→        if (!headerRaw) {
   851→          reject(new Error(`curl 无响应${stderr ? `: ${stderr.slice(0, 160)}` : ''}`))
   852→          return
   853→        }
   854→        // 头文件可能含多轮重定向响应: 按状态行切分逐轮解析 ——
   855→        // 1) 最终状态码/Content-Type 取最后一轮; 2) 每轮 Set-Cookie 归属到该轮实际 URL 的域
   856→        // (经 Location 链逐轮解析)。原先所有轮次的 Cookie 全记在初始 URL 域键下,
   857→        // 跨域/http→https 重定向时会把 B 域 Cookie 发给 A 域(串味+跨站泄漏)
   858→        // ab-b: retryAfter 随轮解析(最终响应轮的 Retry-After 头, 供 429 抛错对象抢救, 同 fetchHttp 口径)
   859→        type CurlRound = { status: number; location: string; contentType: string; setCookies: string[]; retryAfter: string }
   860→        const rounds: CurlRound[] = []
   861→        let cur: CurlRound | null = null
   862→        for (const line of headerRaw.split(/\r?\n/)) {
   863→          const sm = line.match(/^HTTP\/[\d.]+\s+(\d{3})/i)
   864→          if (sm) { cur = { status: parseInt(sm[1], 10), location: '', contentType: '', setCookies: [], retryAfter: '' }; rounds.push(cur); continue }
   865→          if (!cur) continue
   866→          const idx = line.indexOf(':')
   867→          if (idx <= 0) continue
   868→          const key = line.slice(0, idx).trim().toLowerCase()
   869→          const val = line.slice(idx + 1).trim()
   870→          if (key === 'content-type') cur.contentType = val
   871→          else if (key === 'set-cookie') cur.setCookies.push(val)
   872→          else if (key === 'location') cur.location = val
   873→          else if (key === 'retry-after') cur.retryAfter = val // ab-b
   874→        }
   875→        let roundUrl = url
   876→        let status = 0
   877→        let contentType = ''
   878→        let retryAfter = '' // ab-b: 最终响应轮的 Retry-After 原始值
   879→        for (const r of rounds) {
   880→          if (cfg.autoCookie !== false && r.setCookies.length) {
   881→            cookieJar.store(originHost(roundUrl), r.setCookies)
   882→          }
   883→          status = r.status
   884→          contentType = r.contentType
   885→          retryAfter = r.retryAfter
   886→          if (r.status >= 300 && r.status < 400 && r.location) {
   887→            try { roundUrl = new URL(r.location, roundUrl).toString() } catch { /* 非法 Location: 域键保持不变 */ }
   888→          }
   889→        }
   890→        if (status >= 400) {
   891→          const err: any = new Error(`HTTP ${status}(curl)`)
   892→          err.status = status
   893→          err.bodyHtml = decodeBuffer(toArrayBufferView(body), contentType)
   894→          // ab-b: curl 错误形态同样抢救 Retry-After(缺省/非法不挂字段 → 上层 30s 兜底)
   895→          const ram = parseRetryAfterHeaderMs(retryAfter)
   896→          if (ram !== undefined) err.retryAfterMs = ram
   897→          reject(err)
   898→          return
   899→        }
   900→        if (!body.length) {
   901→          reject(new Error(`curl 响应体为空${stderr ? `: ${stderr.slice(0, 160)}` : ''}`))
   902→          return
   903→        }
   904→        resolve(decodeBuffer(toArrayBufferView(body), contentType))
   905→      })().catch(reject)
   906→    })
   907→  })
   908→}

   910→// ---------- bun 中继桥 (gg: node 运行时+代理的 TLS 指纹出路) ----------
   911→/**
   912→ * 场景(gg-b wanben 实录): node 运行时(next dev)下 RequestInit.proxy 被全局 fetch(undici)
   913→ * 静默忽略, 引擎因此直入 curl 链(fetchHttpWithCurlSingle node+proxy 分支), 而 curl 的
   914→ * OpenSSL TLS 指纹被部分 WAF 按 JA3 拦截(同一代理同一 URL: bun 直抓 200 / 引擎 curl 链
   915→ * 403 交替实录), 代理链在 dev 服务里全灭。
   916→ * 中继桥: 独立 bun mini-service(127.0.0.1:3011, Bun 运行时 RequestInit.proxy 原生支持)
   917→ * 代为发起请求, 引擎侧把响应重组为 Response 形态嵌入 fetchHttp 逐跳循环 → node+代理
   918→ * 场景获得 bun 级 TLS 指纹, 逐跳重定向/Cookie/超时/指纹头组语义零改动。
   919→ * 可用性: /health 探测结果缓存 RELAY_PROBE_RETRY_MS; 中继不在/中继层失败 → 原 curl 链
   920→ * 兜底(零回归); 目标侧响应(含 403/5xx)如实上抛不双发。
   921→ * socks5 形态: bun RequestInit.proxy 不支持 → 中继报 relayError → 落 curl(-x 全形态),
   922→ * 与 native 链"socks5 即时失败后 curl 重试"同契约。
   923→ */
   924→/** 中继层错误(区别于目标侧 HTTP 错误): 仅此类错误触发 curl 兜底, 防失败请求双发 */
   925→class RelayTransportError extends Error {
   926→  constructor(message: string) { super(message); this.name = 'RelayTransportError' }
   927→}
   928→
   929→const RELAY_URL = process.env.FETCH_RELAY_URL || 'http://127.0.0.1:3011'
   930→const RELAY_PROBE_RETRY_MS = 60_000
   931→let relayAvailable: boolean | null = null
   932→let relayCheckedAt = 0
   933→
   934→/** 中继可用性探测(/health), 结果按 RELAY_PROBE_RETRY_MS 缓存; 失败短超时快返 */
   935→async function checkRelay(): Promise<boolean> {
   936→  if (relayAvailable === true) return true
   937→  if (relayAvailable === false && Date.now() - relayCheckedAt < RELAY_PROBE_RETRY_MS) return false
   938→  try {
   939→    const res = await fetch(`${RELAY_URL}/health`, { signal: AbortSignal.timeout(1500) })
   940→    relayAvailable = res.ok
   941→  } catch {
   942→    relayAvailable = false
   943→  }
   944→  relayCheckedAt = Date.now()
   945→  return relayAvailable
   946→}
   947→
   948→/** 中继响应重组形态: fetchHttp 逐跳循环消费的最小 Response 面。
   949→ *  body 为可选(rr-c3 卫生: native Response 携带真实流, 中继重组形态无 body 字段 ——
   950→ *  3xx/!ok 分支的 res.body?.cancel() 对中继形态为 no-op, 可选链短路安全) */
   951→interface RelayResponseLike {
   952→  status: number
   953→  ok: boolean
   954→  headers: Headers
   955→  arrayBuffer(): Promise<ArrayBuffer>
   956→  readonly body?: { cancel(): Promise<void> } | null
   957→}
   958→
   959→/** 单跳经中继发起: 返回所有目标侧响应(含 3xx/4xx/5xx, 忠实转发不拦截);
   960→ *  仅中继服务自身不可达/内部错误/代理协议不支持时抛 RelayTransportError。
   961→ *  clientSignal 透传 fetchHttp 的超时控制器(AbortError 原样上抛, 超时分类语义不变);
   962→ *  中继侧超时给 clientTimeoutMs+3000 冗余(客户端先超时, 口径一致) */
   963→async function relayHop(url: string, headers: Headers | Record<string, string>, proxy: string, clientSignal: AbortSignal, clientTimeoutMs: number): Promise<RelayResponseLike> {
   964→  // buildHeaders 返回普通对象, 引擎逐跳处也可能是 Headers 实例 —— 两种形态都收
   965→  const headerObj: Record<string, string> = {}
   966→  if (headers && typeof (headers as Headers).forEach === 'function') {
   967→    ;(headers as Headers).forEach((v, k) => { headerObj[k] = v })
   968→  } else {
   969→    for (const [k, v] of Object.entries(headers as Record<string, string>)) headerObj[k] = v
   970→  }
   971→  let res: Response
   972→  try {
   973→    res = await fetch(`${RELAY_URL}/fetch`, {
   974→      method: 'POST',
   975→      headers: { 'Content-Type': 'application/json' },
   976→      body: JSON.stringify({ url, headers: headerObj, proxy, timeoutMs: clientTimeoutMs + 3000 }),
   977→      signal: clientSignal,
   978→    })
   979→  } catch (e: any) {
   980→    if (e?.name === 'AbortError' || e?.code === 'ABORT_ERR') throw e
   981→    throw new RelayTransportError(`中继不可达(${RELAY_URL}): ${String(e?.message || e).slice(0, 120)}`)
   982→  }
   983→  let payload: { status?: number; headers?: [string, string][]; setCookie?: string[]; bodyB64?: string; relayError?: string }
   984→  try {
   985→    payload = await res.json()
   986→  } catch (e: any) {
   987→    throw new RelayTransportError(`中继响应非 JSON(HTTP ${res.status}): ${String(e?.message || e).slice(0, 100)}`)
   988→  }
   989→  if (payload.relayError || typeof payload.status !== 'number' || typeof payload.bodyB64 !== 'string') {
   990→    throw new RelayTransportError(`中继层失败: ${String(payload.relayError || '响应形态非法').slice(0, 160)}`)
   991→  }
   992→  const h = new Headers()
   993→  for (const [k, v] of payload.headers || []) {
   994→    if (k.toLowerCase() === 'set-cookie') continue // set-cookie 走专用通道保留多条
   995→    try { h.append(k, v) } catch { /* 非法头键跳过(与引擎头键白名单同向) */ }
   996→  }
   997→  for (const c of payload.setCookie || []) {
   998→    try { h.append('set-cookie', c) } catch { /* ignore */ }
   999→  }
  1000→  const buf = Buffer.from(payload.bodyB64, 'base64')
  1001→  return {
  1002→    status: payload.status,
  1003→    ok: payload.status >= 200 && payload.status < 300,
  1004→    headers: h,
  1005→    arrayBuffer: async () => toArrayBufferView(buf),
  1006→  }
  1007→}

  1042→/**
  1043→ * mm-b 反反爬增强: 浏览器类桥模式的 hostGateLimit 自动钳制。
  1044→ * 桥服务对 stealthy/playwright 有全局 BoundedSemaphore(3)(server.py BROWSER_SEM,
  1045→ * 独立 launch 浏览器内存开销大): 引擎侧 hostGateLimit>3 时, 超出部分的请求并不能
  1046→ * 提升采集吞吐, 只会在桥内信号量排队 —— 而"在桥内排队"有两个真实代价:
  1047→ *  ① 白占 hostGate 槽位(同 host 的其他工作/并行任务被 starving);
  1048→ *  ② 引擎客户端护栏(AbortSignal timeout)在排队期间到点后, 请求被放弃但桥内浏览器
  1049→ *     会话继续跑完(孤儿工作), mm 重试还会再叠一次。
  1050→ * 故 gateFetch 准入时把 per-host limit 钳到 min(配置, SCRAPLING_BROWSER_CONCURRENCY):
  1051→ *  - stealthy/playwright → 钳制; 超出部分的请求改走 HostGate 等待(30s 上限),
  1052→ *    超时即 HostGateTimeout —— 系统既有设计面(bb-d: warn/不计失败/章节保持未采集,
  1053→ *    增量重试恢复), 不产生孤儿浏览器工作;
  1054→ *  - static(curl_cffi, 桥内无信号量)/native/未配置 → 原值透传(零回归);
  1055→ *  - 未配置 hostGateLimit(undefined)→ 原样(引擎缺省 3 恰等于钳制线, 无需干预)。
  1056→ * 行为注记: 大 timeout 的 stealthy 配置(桥内排队可在护栏内等到槽位)原先可能"晚到
  1057→ * 成功", 钳制后改为 30s HostGateTimeout + 增量重试 —— 与 hostGate 对超并发的一贯
  1058→ * 语义对齐; pili 生产配置(limit=2)不受影响。
  1059→ * 跨 host 共享: 桥信号量是进程级全局 3, 多 host 并行任务叠加时仍可能在桥内短暂排队
  1060→ * (per-host 闸无法表达全局上限), 该残余排队有界且无害, 不在本钳制职责内。
  1061→ */
  1062→export const SCRAPLING_BROWSER_CONCURRENCY = 3

  1064→export function effectiveHostGateLimit(cfg: Pick<FetchConfig, 'fetchMode' | 'hostGateLimit'>): number | undefined {
  1065→  const mode = scraplingModeOf(cfg.fetchMode)
  1066→  if (mode !== 'stealthy' && mode !== 'playwright') return cfg.hostGateLimit
  1067→  const limit = cfg.hostGateLimit
  1068→  if (typeof limit !== 'number' || !Number.isFinite(limit)) return limit
  1069→  return Math.min(limit, SCRAPLING_BROWSER_CONCURRENCY)
  1070→}

  1072→interface ScraplingBridgeResult {
  1073→  status: number
  1074→  html: string
  1075→  finalUrl: string
  1076→}

  1078→/** 经 scrapling 桥抓取一次: 成功返回 {status,html,finalUrl}; 桥不可达/桥内失败返回 null
  1079→ *  (调用方降级 native 链, warn 已在此打)。目标侧响应(含 4xx/5xx)在 ok:true 信封内如实
  1080→ *  透传 —— 与中继桥契约同向: 仅传输层失败触发降级, 目标请求不双发 */
  1081→async function fetchViaScraplingBridge(url: string, cfg: FetchConfig, mode: ScraplingMode): Promise<ScraplingBridgeResult | null> {
  1082→  const bridge = (cfg.scraplingBridgeUrl || '').trim() || SCRAPLING_BRIDGE_URL
  1083→  // 目标侧代理: 规则配了 proxyUrl 且目标非回环 → 随机选一条交桥(桥内 Fetcher 走代理);
  1084→  // 桥调用本身是回环直连(node fetch 不带 proxy 选项, 与 isLoopbackTarget 豁免语义一致)
  1085→  const proxy = pickProxyFor(url, cfg)
  1086→  // 显式头组透传: 规则 headers 最优先 + cookies 收敛为 Cookie 头(桥内 static 模式可覆盖
  1087→  // 其 stealthy_headers 生成的同名头; stealthy/playwright 经 extra_headers 透传);
  1088→  // native 专有的指纹头组/token 预取头不在此组装(桥内 Fetcher 自生成自洽头组)
  1089→  const headers: Record<string, string> = {}
  1090→  for (const [k, v] of Object.entries(cfg.headers || {})) headers[k] = v
  1091→  if (cfg.cookies && !headers.Cookie) headers.Cookie = cfg.cookies
  1092→  // Referer 与 native buildHeaders 同语义(ii-c 修前补齐: refererChain/refererUrl 运行时注入
  1093→  // 的来源页在 scrapling 模式曾被静默丢失 —— 规则组合 fetchMode=scrapling-* × refererChain
  1094→  // 时目录页→书籍页→章节页 Referer 链断裂): 优先级镜像 buildHeaders = 链 Referer(refererUrl)
  1095→  // > origin 回退(cfg.referer !== false) > 规则显式 cfg.headers.Referer(与 native 一致被回退覆盖)
  1096→  {
  1097→    const chainReferer = cfg.refererChain && cfg.refererUrl ? cfg.refererUrl : ''
  1098→    let origin = ''
  1099→    try { origin = new URL(url).origin } catch { /* ignore */ }
  1100→    if (chainReferer) headers.Referer = chainReferer
  1101→    else if (cfg.referer !== false && origin) headers.Referer = origin
  1102→  }
  1103→  const timeoutMs = cfg.timeout && cfg.timeout > 0 ? cfg.timeout : 20000
  1104→  // mm 轮韧性增强: 桥内浏览器实例逐请求独立, 瞬态崩溃(TargetClosedError/Page crashed,
  1105→  // 多 chromium 并存内存挤压场景)或桥重启窗口, 一次即降级 native —— 对 CF 挑战站 native
  1106→  // 必然 403, 整条采集链当场断裂(生产实锤: pili 任务首页 crash→全任务 0 书)。此处仅对
  1107→  // "桥内失败"重试一次(间隔 800ms, 浏览器重启 typically <1s); 目标侧真实响应(payload.ok)
  1108→  // 永不重发, 与"不双发"契约一致
  1109→  for (let attempt = 1; attempt <= 2; attempt++) {
  1110→    const retryHint = attempt === 1 ? '重试一次' : '降级 native 链'
  1111→    try {
  1112→      const res = await fetch(`${bridge}/fetch`, {
  1113→        method: 'POST',
  1114→        headers: { 'Content-Type': 'application/json' },
  1115→        body: JSON.stringify({ url, mode, headless: true, proxy: proxy || undefined, timeoutMs, headers }),
  1116→        // 客户端侧护栏只防桥进程僵死(桥自身对目标限时); 冗余 15s, 下限 45s(stealthy
  1117→        // 首启含浏览器冷启动 + solve_cloudflare 挑战求解耗时)
  1118→        signal: AbortSignal.timeout(Math.max(timeoutMs + 15_000, 45_000)),
  1119→      })
  1120→      if (!res.ok) {
  1121→        console.warn(`[fetcher] scrapling 桥响应形态非法(HTTP ${res.status}), ${retryHint}`)
  1122→        if (attempt === 1) {
  1123→          await new Promise((r) => setTimeout(r, 800))
  1124→          continue
  1125→        }
  1126→        return null
  1127→      }
  1128→      const payload = (await res.json()) as { ok?: boolean; status?: number; html?: string; finalUrl?: string; error?: string }
  1129→      if (!payload?.ok || typeof payload.status !== 'number' || typeof payload.html !== 'string') {
  1130→        console.warn(`[fetcher] scrapling 桥内失败(${String(payload?.error || '响应形态非法').slice(0, 140)}), ${retryHint}`)
  1131→        if (attempt === 1) {
  1132→          await new Promise((r) => setTimeout(r, 800))
  1133→          continue
  1134→        }
  1135→        return null
  1136→      }
  1137→      return { status: payload.status, html: payload.html, finalUrl: payload.finalUrl || url }
  1138→    } catch (e: any) {
  1139→      console.warn(`[fetcher] scrapling 桥不可达(${bridge}): ${String(e?.message || e).slice(0, 120)}, ${retryHint}`)
  1140→      if (attempt === 1) {
  1141→        await new Promise((r) => setTimeout(r, 800))
  1142→        continue
  1143→      }
  1144→      return null
  1145→    }
  1146→  }
  1147→  return null
  1148→}

  1150→/** 单代理(或直连)单次尝试: bun fetch 失败(网络错误/4xx/5xx)时自动落 curl 子进程。
  1151→ *  超时(AbortError)不落 curl: 同超时下 curl 也救不了, 白等。
  1152→ *  挑战壳(200+JS跳转)不在此处理, 由上层 Cookie 重试/浏览器升级链负责。
  1153→ *  代理尝试在 node 运行时直接走 curl(undici 静默忽略 proxy, 见 PROXY_FETCH_SUPPORTED) */
  1154→async function fetchHttpWithCurlSingle(url: string, cfg: FetchConfig, ua: string, proxy: string): Promise<string> {
  1155→  if (proxy && !PROXY_FETCH_SUPPORTED) {
  1156→    // node 运行时 + 代理: 内置 fetch 不支持 proxy 选项(静默忽略→伪装直连)。
  1157→    // gg 中继桥优先(若 bun 中继服务在位): bun 级 TLS 指纹过 WAF(wanben GoEdge 实录
  1158→    // curl 指纹被拦); 中继不在/中继层失败 → curl 链兜底(既有行为, 零回归)。
  1159→    // 超时(AbortError)不落 curl(同超时 curl 也救不了, 白等); 目标侧响应如实上抛不双发。
  1160→    if (await checkRelay()) {
  1161→      try {
  1162→        return await fetchHttp(url, cfg, ua, proxy, 'relay')
  1163→      } catch (e: any) {
  1164→        if (e?.name === 'AbortError' || e?.code === 'ABORT_ERR') throw e
  1165→        if (!(e instanceof RelayTransportError)) throw e
  1166→        console.warn('[fetcher] 中继桥失败, 落 curl 链:', String(e?.message || e).slice(0, 140))
  1167→      }
  1168→    }
  1169→    return fetchViaCurl(url, cfg, ua, proxy)
  1170→  }
  1171→  try {
  1172→    return await fetchHttp(url, cfg, ua, proxy)
  1173→  } catch (e: any) {
  1174→    if (e?.name === 'AbortError' || e?.code === 'ABORT_ERR') throw e
  1175→    try {
  1176→      return await fetchViaCurl(url, cfg, ua, proxy)
  1177→    } catch (curlErr: any) {
  1178→      console.warn('[fetcher] curl 传输未成:', String(curlErr?.message || curlErr).slice(0, 140))
  1179→      // 原错误是 HTTP 状态错误(带 status)时仍抛原错误保留 bodyHtml 语义;
  1180→      // 原错误是纯网络层失败(无 status, 如 TLS 指纹被 WAF 拒连)时改抛 curl 的错误 ——
  1181→      // 它带 status/bodyHtml, 上层 fetchPage 的 fallbackStatus/Cookie 挑战重试判定依赖这些字段,
  1182→      // 原先一律重抛原错误会让"curl 拿到 403+Set-Cookie"的挑战信号丢失, Cookie 重试链路失效
  1183→      if (e?.status) throw e
  1184→      throw curlErr || e
  1185→    }
  1186→  }
  1187→}

  1196→/**
  1197→ * HTTP 双传输封装 + 出口代理轮换(dd-a, 失败降级契约):
  1198→ * 配置了代理且目标非回环时, Fisher-Yates 洗牌后逐条尝试(每条 = bun fetch→curl 兜底
  1199→ * 单次尝试); 任一条成功即返回; 全部失败 → 降级直连重试一次(与 token 预取
  1200→ * "静默降级不硬断"同口径)。轮换/降级全程仅 warn 级日志, 不因代理失败中断采集;
  1201→ * 降级直连成功与否如实返回/抛出(错误保留 status/bodyHtml 供上层挑战链判定)。
  1202→ * token 预取(prefetchToken)/token 挑战求解(trySolveTokenChallenge)亦经本函数,
  1203→ * 代理/回环豁免语义自动贯穿; 目标回环或未配置代理时行为与原实现完全一致(零回归)
  1204→ */
  1205→export async function fetchHttpWithCurlFallback(url: string, cfg: FetchConfig, ua: string): Promise<string> {
  1206→  const pool = parseProxyPool(cfg.proxyUrl)
  1207→  if (!pool.length || isLoopbackTarget(url)) {
  1208→    return fetchHttpWithCurlSingle(url, cfg, ua, '')
  1209→  }
  1210→  const order = pool.slice()
  1211→  for (let i = order.length - 1; i > 0; i--) {
  1212→    const j = Math.floor(Math.random() * (i + 1))
  1213→    ;[order[i], order[j]] = [order[j], order[i]]
  1214→  }
  1215→  let lastErr: any = null
  1216→  for (const proxy of order) {
  1217→    try {
  1218→      return await fetchHttpWithCurlSingle(url, cfg, ua, proxy)
  1219→    } catch (e: any) {
  1220→      lastErr = e
  1221→      console.warn(`[fetcher] 代理请求失败, 轮换下一条(${redactProxy(proxy)}): ${String(e?.message || e).slice(0, 140)}`)
  1222→    }
  1223→  }
  1224→  console.warn(`[fetcher] 全部 ${order.length} 条代理失败(末次: ${String(lastErr?.message || lastErr).slice(0, 120)}), 降级直连重试: ${url.slice(0, 200)}`)
  1225→  return fetchHttpWithCurlSingle(url, cfg, ua, '')
  1226→}

  1229→/** 场景: 部分站点(bqg713 系 content 段等)的接口/页面需要先从另一端点取得动态 token
  1230→ *  才能放行(缺失/非法值一律 403)。这里提供【通用预取型】token 能力:
  1231→ *   - cfg.tokenUrl: 预取地址(响应体含 token 的任意端点); 支持 {url} 占位符
  1232→ *     (=当前请求 URL 的 encodeURIComponent, 外部转换代理形态)
  1233→ *   - cfg.tokenPattern: 提取表达式 — 'regex:' 前缀=正则(取第一捕获组, 无捕获组取全匹配);
  1234→ *     否则按 JSON 点路径(如 'data.token', 语法同 parser.jsonGet)
  1235→ *   - cfg.tokenInjection: 'url'=替换请求 URL 中的 {token} / %7Btoken%7D 占位符
  1236→ *     (规则 const 模板不认识的 {token} 占位符可用百分号编码形态存活到 fetch 时),
  1237→ *     无占位符时追加 ?token=/&token= 查询参数; 'header'=注入请求头 tokenHeaderName(默认 X-Token)
  1238→ *   - 预取失败/提取为空 → 静默降级为无 token 直连(不硬断链路);
  1239→ *     同 (host+tokenUrl+pattern) 30s 进程内缓存, 防逐章双请求拖慢与预取端限流
  1240→ *   注: bqg713 现状为【按章 AES-CBC 加密参数】型 token(每次请求需对 {id,chapterid}
  1241→ *   加密, 密钥派生自站点混淆 JS), 不属"可预取 token"形态, 本钩子无法表达 ——
  1242→ *   该站仍需站点专属解密或外置转换代理(见 worklog bb-d 留档), 钩子面向通用形态 */
  1243→const TOKEN_CACHE_TTL_MS = 30_000
  1244→/** 容量上限(rr-c3): {url} 占位符形态(生产 bqg713 规则 tokenUrl=…/rewrite?url={url} 在用)
  1245→ *  按解析后 URL 逐章分键, TTL 仅在"同键再次命中"时惰性删过期条目 —— 逐章单次访问的键
  1246→ *  永不回收, 长任务 Map 无界增长(probe-rr-c3-token-cache 实证: 800 目标→800 条目, 人为
  1247→ *  过期条目新写入后仍存活)。超限先全表清扫过期条目, 仍超限按插入序删最旧; 固定 tokenUrl
  1248→ *  (会话型)形态每 host 恒 1 条, 上限永不触及, 缓存语义零变化 */
  1249→const TOKEN_CACHE_MAX = 256
  1250→
  1251→function tokenCacheTrim(cache: Map<string, { token: string; at: number }>) {
  1252→  if (cache.size <= TOKEN_CACHE_MAX) return
  1253→  const now = Date.now()
  1254→  for (const [k, e] of cache) {
  1255→    if (now - e.at >= TOKEN_CACHE_TTL_MS) cache.delete(k)
  1256→  }
  1257→  while (cache.size > TOKEN_CACHE_MAX) {
  1258→    const oldest = cache.keys().next().value
  1259→    if (oldest === undefined) break
  1260→    cache.delete(oldest)
  1261→  }
  1262→}

  1264→const globalForToken = globalThis as unknown as { __novelTokenPrefetch_v1?: Map<string, { token: string; at: number }> }
  1265→function tokenCache(): Map<string, { token: string; at: number }> {
  1266→  if (!globalForToken.__novelTokenPrefetch_v1) globalForToken.__novelTokenPrefetch_v1 = new Map()
  1267→  return globalForToken.__novelTokenPrefetch_v1
  1268→}

  1270→/** token 提取: 'regex:' 前缀=正则第一捕获组(无捕获组取全匹配), 否则 JSON 点路径。
  1271→ *  JSON 路径惰性 import parser(其顶层依赖 fetchPage, 避免模块环) */
  1272→async function extractToken(body: string, pattern: string): Promise<string> {
  1273→  if (!body) return ''
  1274→  const p = (pattern || '').trim()
  1275→  if (!p) return ''
  1276→  if (p.startsWith('regex:')) {
  1277→    try {
  1278→      const m = new RegExp(p.slice(6)).exec(body)
  1279→      return m ? (m[1] ?? m[0] ?? '').trim() : ''
  1280→    } catch { return '' }
  1281→  }
  1282→  try {
  1283→    const { parseJsonBody, jsonGet, jsonToString } = await import('./parser')
  1284→    const root = parseJsonBody(body)
  1285→    if (root === undefined) return ''
  1286→    return jsonToString(jsonGet(root, p)).trim()
  1287→  } catch { return '' }
  1288→}

  1290→/** token 预取(带 30s 进程内缓存): 失败返回 ''(静默降级, 不硬断链路) */
  1291→async function prefetchToken(targetUrl: string, cfg: FetchConfig, ua: string): Promise<string> {
  1292→  const tokenUrl = (cfg.tokenUrl || '').trim()
  1293→  const pattern = (cfg.tokenPattern || '').trim()
  1294→  if (!tokenUrl || !pattern) return ''
  1295→  let real = tokenUrl
  1296→  if (real.includes('{url}')) real = real.replace('{url}', encodeURIComponent(targetUrl))
  1297→  // 缓存键用【解析后】预取 URL(bb-g 修复): 原 tokenUrl 原串含 {url} 占位符时, 同 host 30s 内
  1298→  // 所有目标 URL 共享同一缓存槽 —— 第二章复用第一章的 f(url) token 必被目标端 403。
  1299→  // 固定 tokenUrl(会话型)时 real === tokenUrl, 缓存语义不变
  1300→  const cacheKey = `${originHost(targetUrl)}|${real}|${pattern}`
  1301→  const cached = tokenCache().get(cacheKey)
  1302→  if (cached && Date.now() - cached.at < TOKEN_CACHE_TTL_MS) return cached.token
  1303→  try {
  1304→    const body = await fetchHttpWithCurlFallback(real, cfg, ua)
  1305→    const token = await extractToken(body, pattern)
  1306→    if (token) {
  1307→      const cache = tokenCache()
  1308→      cache.set(cacheKey, { token, at: Date.now() })
  1309→      tokenCacheTrim(cache) // rr-c3: 有界化(修前逐章分键条目永不清扫 → 长任务无界增长)
  1310→    }
  1311→    return token
  1312→  } catch { return '' }
  1313→}

  1315→// ---------- 镜像域名自动故障切换 (dd-b) ----------
  1316→/**
  1317→ * 场景: bqg713 系站点正文 API 钉死单域(apibi.cc), 域死则全站章节全失败, 换模板域名
  1318→ * 需手工平移。引擎级通用能力: FetchConfig.mirrorDomains 配置镜像组, 失败驱动逐 host
  1319→ * 重写重试(transport 级, 本文件内闭环)。
  1320→ * 设计裁定(存档 worklog dd-b):
  1321→ *  - 镜像组 = URL 自身 host + mirrorDomains 全部条目(通用口径: URL host 无须出现在
  1322→ *    列表内, 组内即触发); 组以 URL host 打头, "从当前 host 的下一个开始/环形回绕"语义
  1323→ *    由构造保证(首尝试即当前 host, 逐个后移, 至多组大小次);
  1324→ *  - 触发条件 transport 级: 网络错误/超时(无 status)与 HTTP 403/5xx; 404 与 2xx/3xx
  1325→ *    不触发(404=资源不存在, 换镜像无意义, 存档裁定); 401/412/429 等其余 4xx 亦不触发
  1326→ *    (auto 引擎下交既有浏览器升级链处理, 与镜像职责正交);
  1327→ *  - hostGate 关系: 镜像重试在本层(fetchPage 内)完成, 不经 hostGate 闸门 —— 故障切换是
  1328→ *    失败驱动的低频路径且至多组大小次有界, 按新 host 重新排队会显著复杂化计账且收益为零
  1329→ *    (runner.gateFetch 对整个 fetchPage 调用持一个闸门槽, 内部镜像重试随行同槽);
  1330→ *  - 每个镜像 host 独立走完整 fetchPageOnce 流程(token 预取 {url} 占位符按重写后 URL
  1331→ *    取值 → 逐章 token 天然按镜像域重签, 代理池/回环豁免/UA/Cookie 逻辑照常);
  1332→ *    不做跨请求"上次好域"记忆(有状态缓存会延迟故障发现, 保持无状态可测);
  1333→ *  - fetchBinary(封面等静态资源)刻意不接镜像: 非内容链路且失败优雅降级 null。
  1334→ */
  1335→const MAX_MIRROR_HOSTS = 10
  1336→
  1337→/** 镜像组解析: URL host 打头 + mirrorDomains 逗号分隔条目(小写化/去空/去重/逐条形态
  1338→ *  校验, 上限 MAX_MIRROR_HOSTS; 与 URL host 相同的条目剔除)。未配置或 URL 不可解析
  1339→ *  → 返回空(单 host 直通, 零行为变化) */
  1340→export function mirrorGroupFor(url: string, cfg: Pick<FetchConfig, 'mirrorDomains'>): string[] {
  1341→  const raw = (cfg.mirrorDomains || '').trim()
  1342→  if (!raw) return []
  1343→  let host = ''
  1344→  try { host = new URL(url).host.toLowerCase() } catch { return [] }
  1345→  if (!host) return []
  1346→  const seen = new Set<string>([host])
  1347→  const group = [host]
  1348→  for (const item of raw.split(',')) {
  1349→    const s = item.trim().toLowerCase()
  1350→    if (!s || seen.has(s) || !isValidMirrorHost(s) || group.length >= MAX_MIRROR_HOSTS + 1) continue
  1351→    seen.add(s)
  1352→    group.push(s)
  1353→  }
  1354→  return group
  1355→}

  1357→/** host 重写: 仅换 hostname(条目带 :port 时连 port 一起换, 条目缺省端口则保留原 port),
  1358→ *  scheme/path/query/fragment 原样保留。URL 不可解析返回 null(调用方跳过该镜像) */
  1359→export function rewriteMirrorHost(url: string, hostEntry: string): string | null {
  1360→  try {
  1361→    const u = new URL(url)
  1362→    const idx = hostEntry.lastIndexOf(':')
  1363→    if (idx > 0) {
  1364→      u.hostname = hostEntry.slice(0, idx)
  1365→      u.port = hostEntry.slice(idx + 1)
  1366→    } else {
  1367→      u.hostname = hostEntry
  1368→    }
  1369→    return u.toString()
  1370→  } catch {
  1371→    return null
  1372→  }
  1373→}
  1374→
  1375→/** 镜像切换触发判定: 有显式 status 时仅 403/5xx 可切换; 无 status = 网络层错误/超时
  1376→ *  (DNS 失败/连接拒绝/TLS/AbortError) —— 域名级故障的典型形态, 可切换。
  1377→ *  404(资源不存在, 换镜像无意义, 存档裁定)/3xx/其余 4xx 不触发 */
  1378→export function isMirrorSwitchableError(e: unknown): boolean {
  1379→  const status = (e as { status?: unknown } | null)?.status
  1380→  if (typeof status === 'number' && Number.isFinite(status) && status > 0) {
  1381→    return status === 403 || (status >= 500 && status <= 599)
  1382→  }
  1383→  return true
  1384→}

  1393→/**
  1394→ * 统一抓取入口: 未配置 mirrorDomains 时单 host 直通 fetchPageOnce(与历史行为逐字节一致);
  1395→ * 配置后按镜像组失败驱动切换(dd-b, 语义见镜像段注释)
  1396→ */
  1397→export async function fetchPage(url: string, cfgOverride?: Partial<FetchConfig>): Promise<FetchResult> {
  1398→  const cfg: FetchConfig = { ...DEFAULT_FETCH_CONFIG, ...cfgOverride }
  1399→  const group = mirrorGroupFor(url, cfg)
  1400→  if (group.length <= 1) return fetchPageOnce(url, cfg)
  1401→  let lastErr: unknown = null
  1402→  for (let i = 0; i < group.length; i++) {
  1403→    const hostUrl = rewriteMirrorHost(url, group[i])
  1404→    if (!hostUrl) continue
  1405→    try {
  1406→      return await fetchPageOnce(hostUrl, cfg)
  1407→    } catch (e) {
  1408→      lastErr = e
  1409→      // 不可切换错误(404/3xx/其余4xx)原样上抛: 换镜像无意义, 错误语义与单 host 契约一致
  1410→      if (!isMirrorSwitchableError(e)) throw e
  1411→      console.warn(
  1412→        `[fetcher] 镜像切换: ${group[i]} 失败(${String((e as Error)?.message || e).slice(0, 120)}), ` +
  1413→        (i + 1 < group.length ? `改试下一镜像 ${group[i + 1]}` : `镜像组已尽(共${group.length}个 host)`)
  1414→      )
  1415→    }
  1416→  }
  1417→  throw lastErr ?? new Error('抓取失败(镜像组全部尝试失败)')
  1418→}

  1420→/**
  1421→ * Token 挑战 HTTP 求解器(ixdzs/101kks 系"正在验证浏览器"盾):
  1422→ * 页面 body 内嵌 `let token = "..."` 并执行 `location.href = pathname + "?challenge=" + token`,
  1423→ * 纯 HTTP 即可求解 —— 取 token → 带 Cookie 请求 原地址+challenge 参数 → 得真实页面。
  1424→ * 求解后新 Cookie 已随响应写入 jar, 后续请求直连。
  1425→ * 返回 null 表示不匹配该模式或求解后仍被拦(交回上层升级链)。
  1426→ */
  1427→async function trySolveTokenChallenge(url: string, html: string, cfg: FetchConfig, ua: string): Promise<string | null> {
  1428→  if (!html || html.length > 5000) return null
  1429→  const m = html.match(/token\s*=\s*"([A-Za-z0-9+/=_-]{20,})"/)
  1430→  // 必须同时命中 "?challenge=" 拼接模式才认: 防止 CF "Attention Required" 等无关拦截页
  1431→  // 里的 challenge-platform 字样误触发求解(白烧两跳请求)
  1432→  if (!m || !/["'`]\s*\?\s*challenge\s*=?|challenge\s*=\s*"?["'`+]|\+\s*encodeURIComponent/i.test(html)) return null
  1433→  const challengeUrl = `${url}${url.includes('?') ? '&' : '?'}challenge=${encodeURIComponent(m[1])}`
  1434→  try {
  1435→    const solved = await fetchHttpWithCurlFallback(challengeUrl, cfg, ua)
  1436→    return looksBlocked(solved) ? null : solved
  1437→  } catch {
  1438→    return null
  1439→  }
  1440→}

  1442→/** 单 host 完整抓取流程(原 fetchPage 本体): token 预取 → HTTP 重试链 → auto 浏览器升级。
  1443→ *  每个镜像 host 独立走一遍完整流程 —— token 预取 {url} 占位符按当前 host 的 URL 取值,
  1444→ *  逐章 token 天然按镜像域重签(与 token 钩子组合的正确性来源, verify-dd-b-mirror ④ 实证);
  1445→ *  auto 引擎两条出口错误附加 .status=lastStatus: 镜像层按状态判定可切换性(纯网络错误
  1446→ *  无 status 天然可切换; 404 等不可切换错误透传状态后仍不可切换) */
  1447→async function fetchPageOnce(url: string, cfg: FetchConfig): Promise<FetchResult> {
  1448→  // hh-c: scrapling 桥分流 —— fetchMode='scrapling-*' 时整次抓取交桥代发, 目标侧响应
  1449→  // 如实返回(不双发); native 专有步骤(token 预取/autoCookie/Cookie 重试/浏览器升级链)
  1450→  // 跳过, 隐身能力由桥内 Scrapling Fetcher 承担。桥不可达/桥内异常 → null → 落入下方
  1451→  // 既有 native 链降级一次(warn 日志在 fetchViaScraplingBridge 打出)。非法 fetchMode
  1452→  // 在 sanitize 白名单已丢弃, scraplingModeOf 此处再兜底(运行时对象直改注入防线)。
  1453→  // 未配置 fetchMode / 'native' → scraplingModeOf=null, 下方原流程零行为变化
  1454→  const slMode = scraplingModeOf(cfg.fetchMode)
  1455→  if (slMode) {
  1456→    const bridged = await fetchViaScraplingBridge(url, cfg, slMode)
  1457→    if (bridged) {
  1458→      const blocked = looksBlocked(bridged.html)
  1459→      if (blocked) {
  1460→        console.warn(`[fetcher] scrapling(${slMode}) 内容疑似被拦截(HTTP ${bridged.status}): ${url.slice(0, 160)}`)
  1461→      }
  1462→      return { html: bridged.html, engine: 'http', blocked }
  1463→    }
  1464→  }
  1465→
  1466→  const ua = pickUaFor(originHost(url), cfg)
  1467→
  1468→  // 通用 token 预取(bb-d): tokenUrl+tokenPattern 配置齐全时先取 token, 再按 tokenInjection
  1469→  // 注入('url'=URL 占位符替换/查询参数追加, 'header'=请求头); 未配置/预取失败原样直连
  1470→  let reqUrl = url
  1471→  let effCfg: FetchConfig = cfg
  1472→  if ((cfg.tokenUrl || '').trim() && (cfg.tokenPattern || '').trim()) {
  1473→    const token = await prefetchToken(url, cfg, ua)
  1474→    if (token) {
  1475→      if (cfg.tokenInjection === 'header') {
  1476→        // 请求头名同样清洗控制字符与冒号(与 curl 头注入防护同口径)
  1477→        const name = (cfg.tokenHeaderName || 'X-Token').replace(/[\r\n\0:]+/g, '').trim() || 'X-Token'
  1478→        // 头【值】同样清洗控制字符(bb-g 修复): token 来自远端预取响应体, 值含 CR/LF 时
  1479→        // bun fetch Headers 直接抛 TypeError(硬断链路) / curl 值被空格化、语义破坏 ——
  1480→        // 剥除后仍非空才注入, 否则按预取失败同口径静默降级直连
  1481→        const safeToken = token.replace(/[\x00-\x1f\x7f]+/g, '').trim()
  1482→        if (safeToken) effCfg = { ...cfg, headers: { ...cfg.headers, [name]: safeToken } }
  1483→      } else {
  1484→        const enc = encodeURIComponent(token)
  1485→        if (reqUrl.includes('{token}')) reqUrl = reqUrl.replace('{token}', enc)
  1486→        else if (/%7Btoken%7D/i.test(reqUrl)) reqUrl = reqUrl.replace(/%7Btoken%7D/i, enc)
  1487→        else {
  1488→          // 查询参数追加必须在 #fragment 之前(bb-g 修复): fragment 后的 query 服务端不可见,
  1489→          // 原「直接尾追」会把 token 落进锚点导致目标端永远收不到
  1490→          const h = reqUrl.indexOf('#')
  1491→          const sep = reqUrl.includes('?') ? '&' : '?'
  1492→          reqUrl = h >= 0 ? reqUrl.slice(0, h) + sep + 'token=' + enc + reqUrl.slice(h) : reqUrl + sep + 'token=' + enc
  1493→        }
  1494→      }
  1495→    }
  1496→  }
  1497→
  1498→  const domain = originHost(reqUrl)
  1499→
  1500→  const fallbackStatus = cfg.browserFallbackStatus || [403, 412, 429, 503]
  1501→  let lastErr: any = null
  1502→  let lastStatus = 0
  1503→
  1504→  // 强制浏览器
  1505→  if (cfg.engine === 'browser') {
  1506→    const html = await renderWithBrowser(reqUrl, effCfg, ua)
  1507→    // 浏览器结果同样过拦截判定: 裸 Playwright 降级路径不识别挑战页, 原先固定 blocked=false
  1508→    // 会把盾页当正常内容返回, 上层解析入库产生脏书(obscura 路径已有挑战抛错, 此处是双保险)
  1509→    return { html, engine: 'browser', blocked: looksBlocked(html) }
  1510→  }
  1511→
  1512→  // HTTP 尝试(带重试)
  1513→  // Cookie 挑战重试(修复 guichuideng.info 场景): 首访 403 响应携带 Set-Cookie
  1514→  // (autoCookie 已存入 jar), 带新 Cookie 重发一次 HTTP 即可 200 —— 因此
  1515→  // fallbackStatus 命中时, 若本次响应刚种下新 Cookie 或返回体是 JS 挑战壳,
  1516→  // 不立即 break 升级浏览器, 而是继续下一轮(带新 Cookie), 最多追加 2 次;
  1517→  // 追加用尽仍失败才 break 升级。429/503 仍走原退避重试路径。
  1518→  const baseAttempts = (cfg.retries ?? 0) + 1
  1519→  const MAX_COOKIE_RETRIES = 2
  1520→  let cookieRetries = 0
  1521→  let attempt = 0
  1522→  while (attempt < baseAttempts + cookieRetries) {
  1523→    attempt++
  1524→    const cookiesBefore = cookieJar.count(domain)
  1525→    try {
  1526→      let html = await fetchHttpWithCurlFallback(reqUrl, effCfg, ua)
  1527→      // Token 挑战 HTTP 求解: 命中"正在验证浏览器"式 token 重定向盾时, 纯 HTTP 取 token 重放,
  1528→      // 免浏览器升级(ixdzs/101kks 系)。http 与 auto 引擎均受益
  1529→      if (looksBlocked(html)) {
  1530→        const solved = await trySolveTokenChallenge(reqUrl, html, effCfg, ua)
  1531→        if (solved) html = solved
  1532→      }
  1533→      if (cfg.engine === 'http') return { html, engine: 'http', blocked: looksBlocked(html) }
  1534→      if (!looksBlocked(html)) return { html, engine: 'http', blocked: false }
  1535→      // auto 模式: 200 但内容疑似挑战壳 —— 若刚种下新 Cookie 或响应体是 JS 跳转壳,
  1536→      // 与 403 场景同策略追加带 Cookie 重试(有的站以 200+跳转壳代替 403), 用尽再升级浏览器
  1537→      const gotNewCookieOk = cookieJar.count(domain) > cookiesBefore
  1538→      if ((gotNewCookieOk || isJsChallenge(html)) && cookieRetries < MAX_COOKIE_RETRIES) {
  1539→        cookieRetries++
  1540→        await new Promise((r) => setTimeout(r, 350))
  1541→        continue
  1542→      }
  1543→      lastErr = new Error('内容疑似被拦截(验证码/JS挑战)')
  1544→      break
  1545→    } catch (e: any) {
  1546→      lastErr = e
  1547→      lastStatus = e?.status || 0
  1548→      const bodyHtml: string = e?.bodyHtml || ''
  1549→      // Token 挑战求解(错误路径): 403/412 响应体同样可能是 token 挑战页, 求解成功视同成功
  1550→      if (bodyHtml && looksBlocked(bodyHtml)) {
  1551→        const solved = await trySolveTokenChallenge(reqUrl, bodyHtml, effCfg, ua)
  1552→        if (solved) return { html: solved, engine: 'http', blocked: false }
  1553→      }
  1554→      if (fallbackStatus.includes(lastStatus)) {
  1555→        const gotNewCookie = cookieJar.count(domain) > cookiesBefore
  1556→        if ((gotNewCookie || isJsChallenge(bodyHtml)) && cookieRetries < MAX_COOKIE_RETRIES) {
  1557→          cookieRetries++
  1558→          await new Promise((r) => setTimeout(r, 350))
  1559→          continue // 带刚种下的新 Cookie 重发
  1560→        }
  1561→        // ff-b③: 403 且罐中已有会话但无新 Cookie —— 疑"陈旧会话 Cookie 被目标端拒绝"
  1562→        // (挑战 Cookie 与出口 IP/UA 绑定, 出口轮换后旧罐变毒药)。清空该域罐以全新会话
  1563→        // 重试一次(有 cookieRetries 上限兜底不死循环); 罐为空则直接 break 升级浏览器
  1564→        if (lastStatus === 403 && cookieJar.count(domain) > 0 && cookieRetries < MAX_COOKIE_RETRIES) {
  1565→          cookieJar.clear(domain)
  1566→          cookieRetries++
  1567→          await new Promise((r) => setTimeout(r, 350))
  1568→          continue
  1569→        }
  1570→        // ff-b④: 429/瞬时 5xx(500/502/504) 且重试额度未用尽 → 指数退避重试 HTTP 级
  1571→        // (1.5s×2^n 封顶 8s; 瞬时故障升级浏览器收益低)。retries=0(默认)时 baseAttempts=1
  1572→        // 额度天然耗尽 → 行为与原先完全一致(break 升级), 仅显式配 retries≥1 的规则享受退避。
  1573→        // 503 不参与(常为 CF 挑战壳, 保留升级浏览器语义); 超时不在此路径(isFetchTimeout 另行喂 hostGate)
  1574→        if (
  1575→          (lastStatus === 429 || lastStatus === 500 || lastStatus === 502 || lastStatus === 504) &&
  1576→          attempt < baseAttempts + cookieRetries
  1577→        ) {
  1578→          const delay = Math.min(1500 * Math.pow(2, attempt - 1), 8000)
  1579→          await new Promise((r) => setTimeout(r, delay))
  1580→          continue
  1581→        }
  1582→        break
  1583→      }
  1584→      await new Promise((r) => setTimeout(r, 400 * attempt))
  1585→    }
  1586→  }
  1587→
  1588→  // 本轮 HTTP 全部失败: 释放该域 UA 钉扎, 下次抓取换新 UA 再试
  1589→  domainUa.delete(domain)
  1590→
  1591→  // auto: 升级浏览器渲染
  1592→  if (cfg.engine === 'auto') {
  1593→    const ok = await checkBrowser()
  1594→    if (ok) {
  1595→      try {
  1596→        const html = await renderWithBrowser(reqUrl, effCfg, ua)
  1597→        return { html, engine: 'browser', blocked: looksBlocked(html) }
  1598→      } catch (e: any) {
  1599→        const err: any = new Error(`HTTP(${lastStatus || lastErr?.message}) 与浏览器渲染均失败: ${e?.message?.slice(0, 100)}`)
  1600→        err.status = lastStatus
  1601→        // ab-b: 合成错误透传底层 429 的 Retry-After(gateFetch 抛错路径靠它精确感知限流冷却)
  1602→        if (lastErr && typeof lastErr.retryAfterMs === 'number') err.retryAfterMs = lastErr.retryAfterMs
  1603→        throw err
  1604→      }
  1605→    }
  1606→    const err: any = new Error(
  1607→      `抓取失败(${lastStatus || lastErr?.message || '被拦截'})且浏览器渲染引擎不可用; 可安装chromium或在规则中配置Cookie/UA`
  1608→    )
  1609→    err.status = lastStatus
  1610→    // ab-b: 同上, 合成错误透传 Retry-After(缺失/非法时不挂字段 → 上层 30s 兜底)
  1611→    if (lastErr && typeof lastErr.retryAfterMs === 'number') err.retryAfterMs = lastErr.retryAfterMs
  1612→    throw err
  1613→  }
  1614→  throw lastErr || new Error('抓取失败')
  1615→}

  1617→/** 获取二进制资源(封面等) */
  1618→export async function fetchBinary(
  1619→  url: string,
  1620→  cfgOverride?: Partial<FetchConfig>
  1621→): Promise<{ buf: Buffer; contentType: string } | null> {
  1622→  const cfg: FetchConfig = { ...DEFAULT_FETCH_CONFIG, ...cfgOverride }
  1623→  const ua = pickUaFor(originHost(url), cfg)
  1624→  const headers = buildHeaders(url, cfg, ua)
  1625→  // 大文件内存保护: 封面等资源超过上限直接放弃, 防异常站点回 4GB 响应拖爆内存
  1626→  const MAX_BINARY_BYTES = 25 * 1024 * 1024
  1627→  const timeoutMs = cfg.timeout && cfg.timeout > 0 ? cfg.timeout : 20000
  1628→  const controller = new AbortController()
  1629→  const timer = setTimeout(() => controller.abort(), timeoutMs)
  1630→  try {
  1631→    const res = await fetch(url, { headers, signal: controller.signal, redirect: 'follow' })
  1632→    if (!res.ok) {
  1633→      // rr-c3 卫生: 与 fetchHttp 3xx 分支同口径, 失败路径 body 显式 cancel 释放连接
  1634→      try { void res.body?.cancel().catch(() => {}) } catch { /* ignore */ }
  1635→      return null
  1636→    }
  1637→    const lenHeader = Number(res.headers.get('content-length') || 0)
  1638→    if (lenHeader > MAX_BINARY_BYTES) {
  1639→      // 超限早退时取消响应体: 不消费 body 会占住连接直到服务端断开
  1640→      try { await res.body?.cancel() } catch { /* ignore */ }
  1641→      return null
  1642→    }
  1643→    // 计时器覆盖到响应体读完: 原先 clearTimeout 在 body 读取前, body 传输挂死时无超时兜底
  1644→    const buf = Buffer.from(await res.arrayBuffer())
  1645→    if (buf.length > MAX_BINARY_BYTES) return null
  1646→    return { buf, contentType: res.headers.get('content-type') || '' }
  1647→  } catch {
  1648→    return null
  1649→  } finally {
  1650→    clearTimeout(timer)
  1651→  }
  1652→}
