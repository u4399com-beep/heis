     3→# 小说管理系统 — 零基础一键安装脚本 (中文提示 · 国内网络优化版)

     5→# 做什么: 检测(必要时自动安装) Docker → 自动配置国内镜像加速(可关) →
     6→#         构建镜像 → 启动容器 → 等健康检查通过 → 打印访问地址与自动填充状态。

     9→# 网络自适应(国内服务器友好):
    10→#   - Docker 安装多源自动切换: 官方 get.docker.com → 阿里云 → 清华 → 中科大

   194→probe_url() { # probe_url <url> [connect超时秒, 默认5] → 0=可达
   195→  local url="$1" t="${2:-5}"
   196→  if command -v curl >/dev/null 2>&1; then
   197→    curl -fsS --connect-timeout "$t" --max-time $((t * 3)) -o /dev/null "$url" 2>/dev/null
   198→  elif command -v wget >/dev/null 2>&1; then
   199→    wget -q -T "$t" -t 1 -O /dev/null "$url" 2>/dev/null
   200→  else
   201→    return 2
   202→  fi
   203→}
   204→
   205→git_probe() { # git_probe <超时秒> <仓库url> → 0=可达
   206→  local t="$1" url="$2"
   207→  if command -v timeout >/dev/null 2>&1; then
   208→    timeout "$t" git ls-remote --exit-code "$url" HEAD >/dev/null 2>&1
   209→  else
   210→    git ls-remote --exit-code "$url" HEAD >/dev/null 2>&1
   211→  fi
   212→}

   214→fetch_tarball() { # fetch_tarball <仓库url> <目标目录> — 无 git(或克隆全败)时的压缩包兜底
   215→  # 从 https://github.com/OWNER/REPO 推导归档直链, main → master 依序,
   216→  # 每个分支先试直连再试 GH_PROXY_PREFIXES 加速代理, 下载后解包到目标目录。
   217→  local url="$1" dest="$2" p branch tmp inner
   218→  case "$url" in
   219→    https://github.com/*) ;;
   220→    *) warn "非 GitHub 仓库地址, 无压缩包兜底通道: ${url}"; return 1 ;;
   221→  esac
   222→  local repo_path="${url#https://github.com/}"
   223→  repo_path="${repo_path%.git}"
   224→  for branch in main master; do
   225→    local candidates=("https://github.com/${repo_path}/archive/refs/heads/${branch}.tar.gz")
   226→    for p in "${GH_PROXY_PREFIXES[@]}"; do
   227→      candidates+=("${p}https://github.com/${repo_path}/archive/refs/heads/${branch}.tar.gz")
   228→    done
   229→    local arch
   230→    for arch in "${candidates[@]}"; do
   231→      info "尝试压缩包下载: ${arch}"
   232→      tmp="$(mktemp -d)" || return 1
   233→      if curl -fsSL --connect-timeout 10 --max-time 300 -o "${tmp}/repo.tar.gz" "$arch" 2>/dev/null \
   234→        || wget -q -T 10 -t 1 -O "${tmp}/repo.tar.gz" "$arch" 2>/dev/null; then
   235→        if tar -xzf "${tmp}/repo.tar.gz" -C "$tmp" 2>/dev/null; then
   236→          inner="$(find "$tmp" -mindepth 1 -maxdepth 1 -type d 2>/dev/null | head -n 1)"
   237→          if [ -n "$inner" ]; then
   238→            mkdir -p "$dest"
   239→            # cp 失败必须兜住(磁盘满/权限/dest 异常): 本函数仅在 if 条件上下文被调用,
   240→            # set -e 不会杀脚本, 不守卫会带着假成功 ok+return 0 继续跑(xx-c 复审实证)
   241→            cp -a "$inner"/. "$dest"/ || { rm -rf "$tmp"; return 1; }
   242→            rm -rf "$tmp"
   243→            ok "压缩包下载并解包完成 → ${dest}"
   244→            return 0
   245→          fi
   246→          warn "压缩包内容异常(无顶层目录)"
   247→        else
   248→          warn "压缩包解压失败(文件可能不完整)"
   249→        fi
   250→      else
   251→        warn "压缩包下载失败, 换下一个源重试..."
   252→      fi
   253→      rm -rf "$tmp"
   254→    done
   255→  done
   256→  return 1
   257→}

   259→probe_registry_host() { # probe_registry_host <host> → 0=可达(任意 HTTP 响应含 401 都算; 000=连不上)
   260→  local h="$1" code="000"
   261→  if command -v curl >/dev/null 2>&1; then
   262→    # 注意: curl 失败时 -w 也会输出 000(vv-a 复审实证), 不能再 `|| echo 000` 追加 ——
   263→    # 否则 code="000000" != "000", 不可达主机被误判可达, pick 永远返回第一个候选,
   264→    # 「全挂→daemon.json 加速器兜底」路径成死代码。改为经赋值退出码归一:
   265→    code="$(curl -sS --connect-timeout 4 --max-time 8 -o /dev/null -w '%{http_code}' "https://${h}/v2/" 2>/dev/null)" || code="000"
   266→  fi
   267→  [ -n "$code" ] && [ "$code" != "000" ]
   268→}

   281→clone_repo() { # clone_repo <仓库地址> <目标目录>
   282→  local url="$1" dest="$2"
   283→  if [ -d "${dest}/.git" ]; then
   284→    if command -v git >/dev/null 2>&1; then
   285→      info "目录已存在, 更新代码: ${dest}"
   286→      git -C "$dest" pull --ff-only || warn "git pull 失败, 使用现有代码继续"
   287→    else
   288→      warn "目录已存在但系统未安装 git, 跳过在线更新, 使用现有代码继续"
   289→      warn "如需在线升级: 先安装 git (Ubuntu/Debian: apt install -y git; CentOS: yum install -y git) 后重跑"
   290→    fi
   291→    return 0
   292→  fi
   293→  if ! command -v git >/dev/null 2>&1; then
   294→    warn "未检测到 git 且目录不是 git 仓库, 改走压缩包方式下载..."
   295→    if fetch_tarball "$url" "$dest"; then return 0; fi
   296→    err "压缩包下载也失败。请检查网络后重试, 或先安装 git 后重跑(脚本会自动装):"
   297→    err "  Ubuntu/Debian: apt install -y git    CentOS/RHEL: yum install -y git"
   298→    return 1
   299→  fi
   300→  if git_probe 10 "$url"; then
   301→    info "克隆仓库 ${url} → ${dest}"
   302→    git clone --depth 1 "$url" "$dest"
   303→    return 0
   304→  fi
   305→  warn "GitHub 直连失败(国内网络常见), 尝试加速代理..."
   306→  local p proxied
   307→  for p in "${GH_PROXY_PREFIXES[@]}"; do
   308→    proxied="${p}${url}"
   309→    if git_probe 12 "$proxied"; then
   310→      info "使用加速源 ${p} 克隆..."
   311→      git clone --depth 1 "$proxied" "$dest" || continue
   312→      # origin 指回官方地址, 便于后续正常 git pull/升级
   313→      git -C "$dest" remote set-url origin "$url" 2>/dev/null || true
   314→      return 0
   315→    fi
   316→  done
   317→  warn "git 克隆全败(直连与加速代理均不可达), 最后尝试压缩包方式..."
   318→  if fetch_tarball "$url" "$dest"; then return 0; fi
   319→  err "仓库获取失败(克隆与压缩包均不可达)。请检查网络后重试, 或:"
   320→  err "  A) 手动下载项目压缩包解压后, 进入目录执行 bash install.sh"
   321→  err "  B) 换可用网络/代理后重跑"
   322→  return 1
   323→}

   365→  else
   366→    err "系统缺少 curl/wget, 无法自动安装 Docker; 请先手动安装 curl 后重跑本脚本"

   371→setup_docker_repo_apt() { # setup_docker_repo_apt <镜像站base>  例 https://mirrors.aliyun.com/docker-ce
   372→  local base="$1"
   373→  info "配置 Docker CE 软件源(${base}, 发行版 ${REPO_ID} ${CODENAME})..."
   374→  $SUDO install -m 0755 -d /etc/apt/keyrings
   375→  # gpg 依赖三层降级(yy 轮): Debian trixie/Ubuntu 最小化系统常缺 gpg ——
   376→  # ① gpg 在场 → 原版 dearmor 路径; ② 缺失 → 先尽力补装 gnupg(系统源通常可用);
   377→  # ③ 补装失败 → 改写 .asc 盔甲密钥(apt ≥ 2.4 原生支持 signed-by .asc, 无需 gpg)
   378→  local keyfile="/etc/apt/keyrings/docker.gpg"
   379→  if ! command -v gpg >/dev/null 2>&1; then
   380→    warn "未检测到 gpg, 尝试自动补装 gnupg..."
   381→    export DEBIAN_FRONTEND=noninteractive
   382→    $SUDO apt-get install -y gnupg >/dev/null 2>&1 \
   383→      || { $SUDO apt-get update -y >/dev/null 2>&1 || true; $SUDO apt-get install -y gnupg >/dev/null 2>&1 || true; }
   384→    if command -v gpg >/dev/null 2>&1; then
   385→      ok "gpg 已就绪"
   386→    else
   387→      warn "gpg 暂不可用, 改用免 gpg 的 .asc 密钥方式(apt 2.4+ 支持; 若后续 apt 报 is not signed, 手动 apt install -y gnupg 后重跑)"
   388→      keyfile="/etc/apt/keyrings/docker.asc"
   389→      curl -fsSL --connect-timeout 10 "${base}/linux/${REPO_ID}/gpg" | $SUDO tee "$keyfile" >/dev/null
   390→    fi
   391→  fi
   392→  if [ "$keyfile" = "/etc/apt/keyrings/docker.gpg" ]; then
   393→    curl -fsSL --connect-timeout 10 "${base}/linux/${REPO_ID}/gpg" \
   394→      | $SUDO gpg --dearmor --yes -o "$keyfile"
   395→  fi
   396→  echo "deb [signed-by=${keyfile}] ${base}/linux/${REPO_ID} ${CODENAME} stable" \
   397→    | $SUDO tee /etc/apt/sources.list.d/docker.list >/dev/null
   398→  export DEBIAN_FRONTEND=noninteractive
   399→  $SUDO apt-get update -y
   400→  $SUDO apt-get install -y docker-ce docker-ce-cli containerd.io docker-buildx-plugin docker-compose-plugin
   401→}

   527→configure_registry_mirrors() {
   528→  if [ "$SKIP_REGISTRY_MIRROR" = "1" ]; then
   529→    info "SKIP_REGISTRY_MIRROR=1, 跳过 Docker Hub 镜像加速器配置"
   530→    return 0
   531→  fi
   532→  if [ "$OS" != "Linux" ]; then
   533→    warn "非 Linux 环境, 跳过加速器配置(Docker Desktop 请在其设置界面添加 registry-mirrors)"
   534→    return 0
   535→  fi
   536→  local -a MIRRORS=() USER_MIRRORS=()
   537→  local m
   538→  if [ -n "$REGISTRY_MIRRORS" ]; then
   539→    IFS=',' read -ra USER_MIRRORS <<< "$REGISTRY_MIRRORS"
   540→    # 修剪空白/空项(如 "a,,b" 或 "a, b"); 全空回落默认清单
   541→    # (防 first 为空 → grep -qF "" 对任意文件恒真 → 误判已配置而跳过真实写入)
   542→    for m in "${USER_MIRRORS[@]+"${USER_MIRRORS[@]}"}"; do
   543→      m="$(printf '%s' "$m" | tr -d '[:space:]')"
   544→      if [ -n "$m" ]; then MIRRORS+=("$m"); fi
   545→    done
   546→    if [ "${#MIRRORS[@]}" -eq 0 ]; then
   547→      warn "REGISTRY_MIRRORS 未含有效地址, 回落默认加速器清单"
   548→      MIRRORS=("${REGISTRY_MIRROR_DEFAULTS[@]}")
   549→    fi
   550→  else
   551→    MIRRORS=("${REGISTRY_MIRROR_DEFAULTS[@]}")
   552→  fi
   553→  local first="${MIRRORS[0]}"
   554→  if [ -f /etc/docker/daemon.json ] && grep -qF "$first" /etc/docker/daemon.json 2>/dev/null; then
   555→    ok "Docker Hub 镜像加速器已配置(${first}), 跳过"
   556→    MIRROR_INFO="$first"
   557→    verify_mirrors_active
   558→    return 0
   559→  fi
   560→  local merged=0
   561→  if command -v python3 >/dev/null 2>&1; then
   562→    # 合并写入(保留 daemon.json 已有配置, 不覆盖用户自定义项)
   563→    if $SUDO python3 - "${MIRRORS[@]}" <<'PYEOF'
   564→import json, sys
   565→p = "/etc/docker/daemon.json"
   566→mirrors = [m.strip() for m in sys.argv[1:] if m.strip()]
   567→try:
   568→    with open(p) as f:
   569→        cfg = json.load(f)
   570→except Exception:
   571→    cfg = {}
   572→old = cfg.get("registry-mirrors") or []
   573→merged = mirrors + [m for m in old if m not in mirrors]
   574→cfg["registry-mirrors"] = merged
   575→with open(p, "w") as f:
   576→    json.dump(cfg, f, indent=2, ensure_ascii=False)
   577→print("registry-mirrors =", merged)
   578→PYEOF
   579→    then merged=1; else warn "daemon.json 写入失败(权限/只读盘?), 跳过加速器配置(不影响基础镜像拉取: 已直指镜像站)"; fi
   580→  elif [ ! -s /etc/docker/daemon.json ]; then
   581→    # 无 python3 且无既有配置: 直接写最小 JSON
   582→    # (vv 收官轮修复: 原 IFS='","' 拼接实际以 IFS 首字符 '"' 连接, 产出
   583→    #  ["https://a"https://b"] 非法 JSON → daemon 可能拒启; 改为逐项补 ',' 连接)
   584→    local joint=""
   585→    for m in "${MIRRORS[@]}"; do joint+="${joint:+\",\"}${m}"; done
   586→    echo "{\"registry-mirrors\":[\"${joint}\"]}" | $SUDO tee /etc/docker/daemon.json >/dev/null
   587→    merged=1
   588→  else
   589→    warn "/etc/docker/daemon.json 已存在且系统无 python3, 无法安全合并, 请手动配置后重启 docker:"
   590→    warn "  例: {\"registry-mirrors\": [\"${first}\"]}   然后: sudo systemctl restart docker"
   591→    return 0
   592→  fi
   593→  if [ "$merged" -eq 1 ]; then
   594→    MIRROR_INFO="${MIRRORS[*]}"
   595→    ok "已写入 Docker Hub 镜像加速器(共 ${#MIRRORS[@]} 个, 首选 ${first})"
   596→    info "重启 docker 服务使加速器生效(正在运行的容器会随之重启, 数据不受影响)..."
   597→    if command -v systemctl >/dev/null 2>&1; then
   598→      $SUDO systemctl daemon-reload >/dev/null 2>&1 || true
   599→      $SUDO systemctl restart docker >/dev/null 2>&1 || $SUDO service docker restart >/dev/null 2>&1 \
   600→        || warn "docker 重启失败, 请手动执行: sudo systemctl restart docker"
   601→    else
   602→      $SUDO service docker restart >/dev/null 2>&1 || warn "docker 重启失败, 请手动重启"
   603→    fi
   604→    if "${DOCKER[@]}" info >/dev/null 2>&1; then
   605→      ok "docker 已带加速器重启"
   606→      verify_mirrors_active
   607→    else
   608→      warn "重启后 docker 暂不可用, 请检查: sudo systemctl status docker"
   609→    fi
   610→  fi
   611→}

   624→# ---------- 基础镜像预拉兜底(vv 收官轮) ----------
   625→# 为何预拉: ①构建期 BuildKit 拉取失败时错误深埋 build 输出, 预拉把失败提前暴露并给中文提示
   626→#           ②/v2/ 探活通 ≠ manifest 代理真通(实测存在此形态), 预拉失败自动换站重试并反写
   627→#             BUN_IMAGE/NODE_IMAGE(拉成功的站=构建用站, 经 BUILD_ENV/sudo env 透传 compose)
   628→#           ③预拉成功即入本地存储, 构建期 BuildKit 对 FROM 直接命中本地缓存零网络
   629→# 仅预拉默认构建必需的 BUN_IMAGE/NODE_IMAGE 两个; PYTHON_IMAGE 属可选 scrapling 桥
   630→# (--profile stealthy 才构建), 保持构建参数注入即可, 不预拉省默认安装流量
   631→prepull_base_images() {
   632→  local vn img host rest cand alt switched
   633→  for vn in BUN_IMAGE NODE_IMAGE; do
   634→    img="${!vn}"
   635→    if [ -z "$img" ]; then continue; fi
   636→    if "${DOCKER[@]}" image inspect "$img" >/dev/null 2>&1; then
   637→      info "基础镜像已在本地, 跳过预拉: ${img}"
   638→      continue
   639→    fi
   640→    host="${img%%/*}"; rest="${img#*/}"
   641→    if [ "$rest" = "$img" ]; then
   642→      host=""                      # 无斜杠 = 官方短名(tag 冒号并非 host:port, 先行排除再 case,
   643→    else                           # 否则 node:22-slim 会被 *:* 误判成镜像站前缀, 枉试换站)
   644→      case "$host" in
   645→        *.*|*:*) : ;;               # 含点或端口 = 镜像站 host 前缀(可换站重试)
   646→        *) host=""; rest="$img" ;;  # 官方命名空间短名(如 oven/bun:1) → 无换站语义, 交由构建期拉取
   647→      esac
   648→    fi
   649→    if [ -z "$host" ]; then
   650→      info "基础镜像 ${img} 为官方名, 构建期直接拉取"
   651→      continue
   652→    fi
   653→    info "预拉基础镜像: ${img} ..."
   654→    if "${DOCKER[@]}" pull --quiet "$img" >/dev/null 2>&1; then
   655→      ok "基础镜像预拉成功: ${img} (构建期直接命中本地)"
   656→      continue
   657→    fi
   658→    warn "预拉失败(${img}), 自动尝试其他镜像站..."
   659→    switched=""
   660→    for cand in "${REGISTRY_HOST_CANDIDATES[@]}"; do
   661→      if [ "$cand" = "$host" ]; then continue; fi
   662→      alt="${cand}/${rest}"
   663→      if "${DOCKER[@]}" pull --quiet "$alt" >/dev/null 2>&1; then
   664→        printf -v "$vn" '%s' "$alt"
   665→        export "$vn"
   666→        switched=1
   667→        ok "已换站预拉成功: ${alt} (构建将改用该站)"
   668→        break
   669→      fi
   670→    done
   671→    if [ -z "$switched" ]; then
   672→      warn "各镜像站预拉均失败, 交由构建期继续(若仍卡在拉取, 请手动指定后重跑:"
   673→      warn "  BUN_IMAGE=<镜像站>/oven/bun:1 NODE_IMAGE=<镜像站>/library/node:22-slim bash install.sh)"
   674→    fi
   675→  done
   676→  # 兜底2: 国内模式但未探测到任何镜像站(镜像变量全空) → 用 docker pull 预拉官方名。
   677→  # docker pull 走 daemon 的 registry-mirrors 加速, 与 BuildKit 拉取路径不同(daemon.json
   678→  # 加速器对 BuildKit 兼容不稳正是本轮根因); 官方名镜像一旦进本地存储, 构建期 FROM 直接命中
   679→  if [ "$CN_MODE" = "1" ] && [ -z "$BUN_IMAGE" ] && [ -z "$NODE_IMAGE" ]; then
   680→    local canon
   681→    for canon in oven/bun:1 node:22-slim; do
   682→      if "${DOCKER[@]}" image inspect "$canon" >/dev/null 2>&1; then
   683→        info "基础镜像已在本地, 跳过预拉: ${canon}"
   684→        continue
   685→      fi
   686→      info "预拉基础镜像(经 daemon 加速器): ${canon} ..."
   687→      if "${DOCKER[@]}" pull --quiet "$canon" >/dev/null 2>&1; then

   746→  exit 1

   749→# ── sudo 环境透传(vv-a 复审修复): sudo 默认 env_reset 会剥离本脚本导出(以及用户手动
   750→# export)的构建源/基础镜像/自动填充变量 —— 当 docker 需经 sudo 访问时(非 root 且未入
   751→# docker 组), compose 插值读不到这些变量, 国内模式注入会整体静默失效(基础镜像退回
   752→# docker.io 官方源, 拉取再次超时)。做法: 把非空变量以 VAR=val 形式插到 sudo 与 docker
   753→# 之间(sudo env VAR=... docker compose), 不受 sudoers 环境策略影响; root/直连路径
   754→# (SUDO 空)export 本身已足够, 此块自动跳过。
   755→BUILD_ENV=()
   756→[ -n "$BUN_IMAGE" ] && BUILD_ENV+=("BUN_IMAGE=$BUN_IMAGE")
   757→[ -n "$NODE_IMAGE" ] && BUILD_ENV+=("NODE_IMAGE=$NODE_IMAGE")
   758→[ -n "$PYTHON_IMAGE" ] && BUILD_ENV+=("PYTHON_IMAGE=$PYTHON_IMAGE")
   759→[ -n "$NPM_REGISTRY" ] && BUILD_ENV+=("NPM_REGISTRY=$NPM_REGISTRY")
   760→[ -n "$PIP_INDEX_URL" ] && BUILD_ENV+=("PIP_INDEX_URL=$PIP_INDEX_URL")
   761→[ -n "$DEBIAN_MIRROR" ] && BUILD_ENV+=("DEBIAN_MIRROR=$DEBIAN_MIRROR")
   762→[ -n "$PLAYWRIGHT_DOWNLOAD_HOST" ] && BUILD_ENV+=("PLAYWRIGHT_DOWNLOAD_HOST=$PLAYWRIGHT_DOWNLOAD_HOST")
   763→[ -n "$AUTO_FILL" ] && BUILD_ENV+=("AUTO_FILL=$AUTO_FILL")
   764→[ -n "$AUTO_FILL_RULES" ] && BUILD_ENV+=("AUTO_FILL_RULES=$AUTO_FILL_RULES")
   765→if [ "${#BUILD_ENV[@]}" -gt 0 ] && [ -n "$SUDO" ] && [ "${COMPOSE[0]}" = "$SUDO" ]; then
   766→  COMPOSE=("$SUDO" env "${BUILD_ENV[@]}" "${COMPOSE[@]:1}")
   767→  info "已将 ${#BUILD_ENV[@]} 个构建变量经 sudo env 透传给 compose(不受 sudo 环境重置影响)"
   768→fi
